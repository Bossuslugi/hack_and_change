package main

func main() {
        if len(os.Args) > 1 && (os.Args[1] == "-healthcheck" || os.Args[1] == "--healthcheck") {
                port := os.Getenv("PORT")
                if port == "" {
                        port = "8080"
                }
                resp, err := http.Get("http://127.0.0.1:" + port + "/healthz")
                if err != nil || resp.StatusCode != http.StatusOK {
                        os.Exit(1)
                }
                os.Exit(0)
        }
        log := slog.New(slog.NewJSONHandler(os.Stdout, nil))
        ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
        defer stop()
        if err := run(ctx, config.Load(os.Getenv), log, nil); err != nil {
                log.Error("сервер остановлен с ошибкой", "error", err.Error())
                os.Exit(1)
        }
}

// run собирает приложение и обслуживает HTTP до отмены контекста (graceful shutdown).
// ready, если задан, получает фактический адрес после начала прослушивания.
func run(ctx context.Context, cfg config.Config, log *slog.Logger, ready chan<- string) error {
        application, err := app.Build(ctx, cfg, log, app.Overrides{})
        if err != nil {
                return fmt.Errorf("не удалось запустить приложение: %w", err)
        }
        defer application.Close()

        listener, err := net.Listen("tcp", ":"+cfg.Port)
        if err != nil {
                return fmt.Errorf("не удалось занять порт %s: %w", cfg.Port, err)
        }
        srv := &http.Server{Handler: application.Handler, ReadHeaderTimeout: 5 * time.Second}

        stopped := make(chan error, 1)
        go func() {
                <-ctx.Done()
                shutdownCtx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
                defer cancel()
                stopped <- srv.Shutdown(shutdownCtx)
        }()

        log.Info("AI-Buddy backend запущен", "addr", listener.Addr().String())
        if ready != nil {
                ready <- listener.Addr().String()
        }
        if err := srv.Serve(listener); err != nil && !errors.Is(err, http.ErrServerClosed) {
                return err
        }
        return <-stopped
}

// Stage — этап обработки запроса (для статуса ожидания в интерфейсе).

// Engine — оркестратор агента.
type Engine struct {
        spec    *spec.AgentSpec
        tools   ToolExecutor
        catalog ToolCatalog
        flags   FlagReader
        llm     LLM
        clock   domain.Clock
        delay   time.Duration
        sleep   func(ctx context.Context, d time.Duration)
}

// Options — параметры оркестратора.
type Options struct {
        Spec    *spec.AgentSpec
        Tools   ToolExecutor
        Catalog ToolCatalog
        Flags   FlagReader
        LLM     LLM // nil — только Smart Engine
        Clock   domain.Clock
        // Delay — суммарная имитация «печатает…» Smart Engine; <0 — взять из спецификации.
        Delay time.Duration
}

// New создаёт оркестратор.
func New(o Options) *Engine {
        delay := o.Delay
        if delay < 0 {
                delay = time.Duration(o.Spec.Agent.Engine.SmartDelayMS) * time.Millisecond
        }
        // Интерфейс с nil-указателем внутри не равен nil — нормализуем.
        if c, ok := o.LLM.(*OpenAIClient); ok && c == nil {
                o.LLM = nil
        }
        return &Engine{
                spec: o.Spec, tools: o.Tools, catalog: o.Catalog, flags: o.Flags, llm: o.LLM, clock: o.Clock,
                delay: delay, sleep: sleepCtx,
        }
}

func sleepCtx(ctx context.Context, d time.Duration) {
        if d <= 0 {
                return
        }
        t := time.NewTimer(d)
        defer t.Stop()
        select {
        case <-ctx.Done():
        case <-t.C:
        }
}

// Mode возвращает режим работы и модель для отображения в интерфейсе.
func (e *Engine) Mode(ctx context.Context) (mode, model string) {
        if e.llmEnabled(ctx) {
                return domain.EngineLLM, e.llm.Model()
        }
        return domain.EngineSmart, "Smart Engine (детерминированный)"
}

func (e *Engine) llmEnabled(ctx context.Context) bool {
        return e.llm != nil && e.flags.Enabled(ctx, mcp.FlagLLMAnswers)
}

func (e *Engine) pause(ctx context.Context) { e.sleep(ctx, e.delay/3) }

func emit(req Request, kind, label, tool string) {
        if req.Emit != nil {
                req.Emit(Stage{Kind: kind, Label: label, Tool: tool})
        }
}

// Handle обрабатывает сообщение стажёра.
func (e *Engine) Handle(ctx context.Context, req Request) Reply {
        emit(req, StageUnderstanding, "Разбираюсь в запросе…", "")
        e.pause(ctx)

        intent := Classify(e.spec, req.Text)
        var r Reply
        switch intent.ID {
        case IntentAffirmative:
                intent, r = e.affirmative(ctx, req, intent)
        case IntentRestricted:
                r = e.restricted(ctx, req, intent)
        case IntentAccess:
                r = e.access(ctx, req, intent)
        case IntentAbsence:
                r = e.absence(ctx, req, intent)
        case IntentProcess, IntentTaskHelp:
                r = e.knowledge(ctx, req, intent)
        case IntentNextStep:
                r = e.nextStep(req)
        case IntentMentor:
                r = e.mentorContact(req)
        case IntentGreeting:
                r = e.greeting(req)
        default:
                var ok bool
                if e.llmEnabled(ctx) {
                        r, ok = e.llmAgent(ctx, req)
                }
                if !ok {
                        reason := r.LLMFallback
                        r = e.knowledge(ctx, req, intent)
                        if reason != "" {
                                // LLM уже не справилась в этом ходе: отвечает Smart Engine, повторно её не дёргаем.
                                r.LLMFallback, r.Engine = reason, domain.EngineSmart
                        }
                }
        }

        r.Intent, r.Scenario = intent.ID, intent.Scenario
        if len(r.Suggestions) == 0 {
                r.Suggestions = append([]string{}, intent.Suggestions...)
        }
        if r.Verdict == nil {
                r.Verdict = pickVerdict(r.ToolCalls)
        }
        if r.Engine == "" {
                r.Engine = domain.EngineSmart
                if e.llmEnabled(ctx) && rephrasable(intent.ID) && r.Answered {
                        e.rephrase(ctx, req, &r)
                }
        }
        emit(req, StageComposing, "Формулирую ответ…", "")
        if r.Engine == domain.EngineSmart {
                e.pause(ctx)
        }
        return r
}

// pickVerdict выбирает итоговый вердикт RBAC: первый запрет, иначе вердикт последнего вызова.
func pickVerdict(calls []domain.MCPToolCall) *domain.RBACVerdict {
        if len(calls) == 0 {
                return nil
        }
        for i := range calls {
                if !calls[i].RBAC.Allowed {
                        v := calls[i].RBAC
                        return &v
                }
        }
        v := calls[len(calls)-1].RBAC
        return &v
}

// rephrasable — интенты, ответы на которые можно отдать LLM на переформулировку.
// Отказ по безопасности (restricted_data_export) всегда детерминирован.
func rephrasable(intent string) bool {
        switch intent {
        case IntentProcess, IntentTaskHelp, IntentKnowledge, IntentAbsence, IntentAccess, IntentNextStep, IntentMentor:
                return true
        }
        return false
}

func (e *Engine) call(ctx context.Context, req Request, tool string, args map[string]any, label string) domain.MCPToolCall {
        emit(req, StageRBAC, fmt.Sprintf("Проверяю права роли %s…", req.Principal.Role), tool)
        emit(req, StageTool, label, tool)
        e.pause(ctx)
        return e.tools.Execute(ctx, req.Principal, req.Intern.ID, tool, args)
}

func deniedCard(v domain.RBACVerdict) *domain.ResultCard {
        card := &domain.ResultCard{
                Type: domain.CardRBACDenied, Status: domain.CardStatusDenied,
                Title: "Доступ запрещён · FORBIDDEN (403)",
                Lines: []domain.CardLine{
                        {Label: "Операция", Value: v.Operation},
                        {Label: "Роль", Value: string(v.Role)},
                        {Label: "Политика", Value: v.PolicyID},
                        {Label: "Причина", Value: v.Reason},
                },
        }
        if v.Incident {
                card.Lines = append(card.Lines, domain.CardLine{Label: "Инцидент", Value: "зафиксирован в журнале безопасности"})
        }
        if v.SafeAlternative != "" {
                card.Lines = append(card.Lines, domain.CardLine{Label: "Безопасная альтернатива", Value: v.SafeAlternative})
        }
        return card
}

func (e *Engine) toolFailure(req Request, call domain.MCPToolCall) string {
        if call.Status == domain.ToolInvalid {
                return e.spec.Response("invalid_arguments", map[string]string{"details": call.Error})
        }
        return e.spec.Response("tool_error", map[string]string{"server": call.Server, "mentor": req.Intern.MentorName})
}

func (e *Engine) restricted(ctx context.Context, req Request, intent spec.Intent) Reply {
        dataset := "users_pii"
        if domain.MatchTrigger([]string{"зарплат|ведомост|оклад|salary|payroll"}, req.Text) {
                dataset = "payroll"
        }
        call := e.call(ctx, req, intent.Tool, map[string]any{
                "dataset": dataset, "reason": truncate(req.Text, 500), "requested_by": req.Principal.ID,
        }, "Обращаюсь к db_admin через RBAC-шлюз…")

        r := Reply{ToolCalls: []domain.MCPToolCall{call}, Answered: true, SavingKey: domain.SavingBlocked}
        v := call.RBAC
        if call.Status != domain.ToolForbidden {
                // Недостижимо при корректной матрице; на случай ошибки в спецификации отказываем всё равно.
                v = domain.RBACVerdict{
                        Allowed: false, Code: domain.VerdictForbidden, HTTPStatus: 403, Role: req.Principal.Role,
                        Operation: intent.Tool, PolicyID: "POL-SEC-001", Incident: true,
                        Reason: "Выгрузка production-данных с ПДн и финансовых сведений запрещена",
                }
        }
        r.Verdict = &v
        r.Text = e.spec.Response("rbac_forbidden", map[string]string{
                "operation": v.Operation, "policy": v.PolicyID, "role": string(v.Role),
        }) + "\n\n" + e.spec.Response("rbac_safe_alternative", nil)
        r.ResultCard = deniedCard(v)
        return r
}

func (e *Engine) access(ctx context.Context, req Request, intent spec.Intent) Reply {
        repo := ExtractRepository(req.Text)
        if repo == "" {
                repo = req.Intern.Repository
        }
        stand := "test"
        if domain.MatchTrigger([]string{"staging|стейдж|стейджинг"}, req.Text) {
                stand = "staging"
        }
        call := e.call(ctx, req, intent.Tool, map[string]any{
                "intern_id": req.Intern.ID, "repository": repo, "stand": stand, "access_level": "developer",
                "justification": "Онбординг стажёра: " + req.Intern.ContextLine(),
        }, "Запрашиваю доступ в IAM…")
        r := Reply{ToolCalls: []domain.MCPToolCall{call}, Answered: true}

        switch call.Status {
        case domain.ToolSuccess:
                res := call.Result
                standName, _ := res["stand"].(string)
                token, _ := res["token_masked"].(string)
                level, _ := res["access_level"].(string)
                expires := ""
                if s, _ := res["expires_at"].(string); len(s) >= 10 {
                        if t, err := time.Parse("2006-01-02", s[:10]); err == nil {
                                expires = t.Format("02.01.2006")
                        }
                }
                already, _ := res["already_granted"].(bool)
                var b strings.Builder
                if already {
                        fmt.Fprintf(&b, "Доступ к **%s** у тебя уже есть — повторно выдавать не нужно.\n\n", repo)
                } else {
                        fmt.Fprintf(&b, "Готово! Проверил твою роль (%s, команда %s) — доступ разрешён политикой %s, и я оформил его через IAM.\n\n",
                                req.Principal.Role, req.Intern.Team, call.RBAC.PolicyID)
                }
                fmt.Fprintf(&b, "- Репозиторий: **%s**, уровень %s\n", repo, level)
                if standName != "" {
                        fmt.Fprintf(&b, "- Тестовый стенд: **%s** (только синтетические данные)\n", standName)
                }
                fmt.Fprintf(&b, "- Токен: %s — синтетический, действует до %s", token, expires)
                r.Text = b.String()
                if !already {
                        r.SavingKey = domain.SavingAccess // повторный запрос не экономит время наставника
                }
                r.facts = []string{repo}
                lines := []domain.CardLine{
                        {Label: "Репозиторий", Value: repo},
                        {Label: "Уровень доступа", Value: level},
                }
                if standName != "" {
                        lines = append(lines, domain.CardLine{Label: "Тестовый стенд", Value: standName})
                }
                lines = append(lines,
                        domain.CardLine{Label: "Токен", Value: token},
                        domain.CardLine{Label: "Действует до", Value: expires},
                        domain.CardLine{Label: "RBAC", Value: fmt.Sprintf("%s · %s", call.RBAC.Code, call.RBAC.PolicyID)},
                )
                r.ResultCard = &domain.ResultCard{
                        Type: domain.CardAccessGranted, Status: domain.CardStatusSuccess, Title: "Доступ выдан",
                        Lines: lines, Footnote: "Имитация IAM · синтетические данные",
                }
        case domain.ToolForbidden:
                r.Text = fmt.Sprintf(
                        "Не могу выдать доступ к **%s**: %s (политика %s, FORBIDDEN). "+
                                "Доступ к репозиториям других команд запрашивает руководитель — %s.\n\n"+
                                "Могу оформить доступ к репозиторию твоей команды: **%s**.",
                        repo, call.RBAC.Reason, call.RBAC.PolicyID, req.Intern.TeamLeadName, req.Intern.Repository)
                r.ResultCard = deniedCard(call.RBAC)
                r.Suggestions = []string{fmt.Sprintf("Дай доступ к репозиторию %s и тестовому стенду", req.Intern.Repository)}
        default:
                r.Text = e.toolFailure(req, call)
                r.Answered = false
        }
        return r
}

func (e *Engine) search(ctx context.Context, req Request, topics []string) (domain.MCPToolCall, []domain.DocRef) {
        args := map[string]any{
                "query": truncate(req.Text, 500), "team": req.Intern.Team, "stack": req.Intern.Stack,
                "stage": req.Intern.Stage, "limit": 3,
        }
        if len(topics) > 0 {
                list := make([]any, 0, len(topics))
                for _, t := range topics {
                        list = append(list, t)
                }
                args["topics"] = list
        }
        call := e.call(ctx, req, mcp.ToolSearchDocs, args, "Ищу в базе знаний команды…")
        if !call.OK() {
                return call, nil
        }
        return call, mcp.DocRefs(call.Result)
}

func writeDoc(b *strings.Builder, d domain.DocRef) {
        fmt.Fprintf(b, "**%s**\n", d.Title)
        if d.Summary != "" {
                b.WriteString(d.Summary + "\n")
        }
        for i, step := range d.Steps {
                fmt.Fprintf(b, "%d. %s\n", i+1, step)
        }
        b.WriteString("\n")
}

func docIDs(docs []domain.DocRef) []string {
        ids := make([]string, 0, len(docs))
        for _, d := range docs {
                ids = append(ids, d.ID)
        }
        return ids
}

func (e *Engine) knowledge(ctx context.Context, req Request, intent spec.Intent) Reply {
        call, docs := e.search(ctx, req, intent.Topics)
        r := Reply{ToolCalls: []domain.MCPToolCall{call}}
        if !call.OK() {
                r.Text = e.toolFailure(req, call)
                return r
        }
        if len(docs) == 0 {
                r.Text = e.spec.Response("not_found", map[string]string{"mentor": req.Intern.MentorName})
                r.OpenQuestion = truncate(req.Text, 300)
                return r
        }

        var b strings.Builder
        fmt.Fprintf(&b, "Учитываю твой контекст: %s.\n\n", req.Intern.ContextLine())
        if intent.ID == IntentTaskHelp && req.Task.Key != "" {
                fmt.Fprintf(&b, "Твоя задача **%s**: %s.\n\n", req.Task.Key, req.Task.Summary)
        }
        for _, d := range docs {
                writeDoc(&b, d)
        }
        if strings.HasPrefix(req.Intern.Stage, "Week 1") {
                switch intent.ID {
                case IntentProcess:
                        fmt.Fprintf(&b, "Для первой недели: первую выкладку и включение флага делай в паре с наставником (%s) — так принято в команде %s.\n\n",
                                req.Intern.MentorName, req.Intern.Team)
                case IntentTaskHelp:
                        fmt.Fprintf(&b, "Если после этого останется архитектурный вопрос — я добавлю его в бриф к 1:1 с наставником (%s).\n\n", req.Intern.MentorName)
                }
        }
        b.WriteString("Источники — тестовые инструкции (синтетические данные), а не регламенты Авито.")

        r.Text = b.String()
        r.Sources = docs
        r.DocsRead = docIDs(docs)
        r.Answered = true
        r.SavingKey = domain.SavingQuestion
        return r
}

func (e *Engine) absence(ctx context.Context, req Request, intent spec.Intent) Reply {
        call, docs := e.search(ctx, req, intent.Topics)
        r := Reply{ToolCalls: []domain.MCPToolCall{call}, Sources: docs, DocsRead: docIDs(docs), Answered: true, SavingKey: domain.SavingQuestion}

        var b strings.Builder
        fmt.Fprintf(&b, "Коротко: отдельно писать руководителю не нужно — заявка в сервисе «Отсутствия» сама уведомит наставника (%s) и руководителя (%s). "+
                "Если из-за отгула пропускаешь встречу — предупреди наставника в командном чате.\n\n", req.Intern.MentorName, req.Intern.TeamLeadName)
        if len(docs) > 0 {
                fmt.Fprintf(&b, "**Порядок в команде %s:**\n", req.Intern.Team)
                for i, step := range docs[0].Steps {
                        fmt.Fprintf(&b, "%d. %s\n", i+1, step)
                }
                b.WriteString("\n")
        }

        if !e.flags.Enabled(ctx, mcp.FlagAbsenceCard) {
                b.WriteString("Интерактивная карточка оформления сейчас выключена feature-флагом — оформи заявку в сервисе «Отсутствия» по шагам выше.")
                r.Text = b.String()
                return r
        }

        draft := ParseAbsence(req.Text, e.clock.Now())
        if draft.DateFound {
                fmt.Fprintf(&b, "Я подготовил заявку: **%s, %s–%s**, причина — «%s». ", HumanDay(draft.Date), draft.TimeFrom, draft.TimeTo, draft.Reason)
                r.facts = []string{draft.TimeFrom}
        } else {
                b.WriteString("Я подготовил заявку, но не понял дату — укажи её в карточке. ")
        }
        b.WriteString("Проверь данные и нажми «Оформить отгул»: без твоего подтверждения я ничего не отправляю.")
        r.Text = b.String()
        r.ActionCard = &domain.ActionCard{
                Type: domain.ActionAbsenceForm, Title: "Заявка на отгул",
                Description: "Сервис «Отсутствия» (имитация) · после подтверждения уведомлю наставника и руководителя",
                Tool:        intent.ConfirmTool,
                Fields: map[string]any{
                        "intern_id": req.Intern.ID, "date": draft.Date, "time_from": draft.TimeFrom, "time_to": draft.TimeTo,
                        "reason": draft.Reason, "notify": []any{"mentor", "team_lead"},
                },
                SubmitLabel: "Оформить отгул", Status: domain.ActionStatusPending,
        }
        return r
}

// rephrase просит LLM переформулировать детерминированный черновик, не меняя фактов.
// При любой ошибке или потере фактов остаётся черновик Smart Engine.
func (e *Engine) rephrase(ctx context.Context, req Request, r *Reply) {
        emit(req, StageLLM, "Формулирую ответ с помощью LLM…", "")
        resp, err := e.llm.Complete(ctx, LLMRequest{Messages: []LLMMessage{
                {Role: "system", Content: e.systemPrompt(req)},
                {Role: "user", Content: fmt.Sprintf(
                        "Запрос стажёра: «%s»\n\nЧерновик ответа, собранный из результатов MCP-инструментов и базы знаний:\n---\n%s\n---\n"+
                                "Перепиши черновик естественным языком. Сохрани ВСЕ факты, шаги, имена, номера, идентификаторы политик и названия "+
                                "репозиториев дословно. Ничего не добавляй от себя. Допустимая разметка: **жирный**, нумерованные и маркированные списки. "+
                                "Верни только текст ответа.", truncate(req.Text, 500), r.Text)},
        }})
        switch {
        case err != nil:
                r.LLMFallback = err.Error()
        case strings.TrimSpace(resp.Content) == "":
                r.LLMFallback = "LLM вернула пустой ответ"
        case !keepsFacts(resp.Content, r.facts):
                r.LLMFallback = "LLM потеряла факты из результата инструмента"
        default:
                r.Text, r.Engine = strings.TrimSpace(resp.Content), domain.EngineLLM
        }
}

func keepsFacts(text string, facts []string) bool {
        for _, f := range facts {
                if f != "" && !strings.Contains(text, f) {
                        return false
                }
        }
        return true
}

const maxLLMRounds = 3

func llmToolName(name string) string { return strings.ReplaceAll(name, ".", "__") }
func mcpToolName(name string) string { return strings.ReplaceAll(name, "__", ".") }
func jsonText(v any) string          { raw, _ := json.Marshal(v); return string(raw) }
func hasProperty(t spec.Tool, key string) bool {
        props, _ := t.InputSchema["properties"].(map[string]any)
        _, ok := props[key]
        return ok
}

// llmAgent — свободный режим для запросов вне известных сценариев: LLM сама выбирает
// MCP-инструменты (function calling). Каждый вызов проходит тот же RBAC-шлюз;
// инструменты с requires_confirmation не исполняются, а превращаются в карточку подтверждения.
func (e *Engine) llmAgent(ctx context.Context, req Request) (Reply, bool) {
        emit(req, StageLLM, "Думаю вместе с LLM…", "")
        var tools []LLMTool
        for _, t := range e.catalog.Tools() {
                tools = append(tools, LLMTool{Type: "function", Function: LLMFunction{
                        Name: llmToolName(t.Name), Description: t.Description, Parameters: t.InputSchema,
                }})
        }
        messages := []LLMMessage{
                {Role: "system", Content: e.systemPrompt(req)},
                {Role: "user", Content: truncate(req.Text, 2000)},
        }
        r := Reply{Engine: domain.EngineLLM}
        for round := 0; round < maxLLMRounds; round++ {
                resp, err := e.llm.Complete(ctx, LLMRequest{Messages: messages, Tools: tools})
                if err != nil {
                        return Reply{LLMFallback: err.Error()}, false
                }
                if len(resp.ToolCalls) == 0 {
                        if strings.TrimSpace(resp.Content) == "" {
                                return Reply{LLMFallback: "LLM вернула пустой ответ"}, false
                        }
                        r.Text = resp.Content
                        break
                }
                messages = append(messages, LLMMessage{Role: "assistant", Content: resp.Content, ToolCalls: resp.ToolCalls})
                for _, tc := range resp.ToolCalls {
                        messages = append(messages, LLMMessage{Role: "tool", ToolCallID: tc.ID, Content: e.runLLMTool(ctx, req, tc, &r)})
                }
        }
        if r.Text == "" {
                return Reply{LLMFallback: "LLM не дала финальный ответ за отведённое число шагов"}, false
        }
        r.Answered = true
        r.SavingKey = domain.SavingQuestion
        if v := pickVerdict(r.ToolCalls); v != nil && !v.Allowed {
                r.ResultCard = deniedCard(*v)
                if v.Incident {
                        r.SavingKey = domain.SavingBlocked
                }
        }
        return r, true
}

func (e *Engine) runLLMTool(ctx context.Context, req Request, tc LLMToolCall, r *Reply) string {
        name := mcpToolName(tc.Function.Name)
        args := map[string]any{}
        if raw := strings.TrimSpace(tc.Function.Arguments); raw != "" {
                if err := json.Unmarshal([]byte(raw), &args); err != nil {
                        return jsonText(map[string]any{"status": "invalid", "error": "аргументы не являются JSON-объектом"})
                }
        }
        toolSpec, known := e.catalog.Tool(name)
        if known && hasProperty(toolSpec, "intern_id") {
                args["intern_id"] = req.Intern.ID // агент действует только от имени текущего стажёра
        }
        if known && toolSpec.RequiresConfirmation {
                if _, ok := args["notify"]; !ok {
                        args["notify"] = []any{"mentor", "team_lead"}
                }
                r.ActionCard = &domain.ActionCard{
                        Type: domain.ActionAbsenceForm, Title: "Заявка на отгул",
                        Description: "Сервис «Отсутствия» (имитация) · после подтверждения уведомлю наставника и руководителя",
                        Tool:        name, Fields: args, SubmitLabel: "Оформить отгул", Status: domain.ActionStatusPending,
                }
                return jsonText(map[string]any{
                        "status": "awaiting_user_confirmation",
                        "note":   "Пользователю показана карточка подтверждения. Действие будет выполнено только после его подтверждения в интерфейсе.",
                })
        }
        call := e.call(ctx, req, name, args, fmt.Sprintf("Вызываю %s…", name))
        r.ToolCalls = append(r.ToolCalls, call)
        if call.OK() && name == mcp.ToolSearchDocs {
                docs := mcp.DocRefs(call.Result)
                r.Sources = append(r.Sources, docs...)
                r.DocsRead = append(r.DocsRead, docIDs(docs)...)
        }
        return jsonText(map[string]any{
                "status": call.Status, "result": call.Result, "error_code": call.ErrorCode, "error": call.Error,
                "rbac": map[string]any{"code": call.RBAC.Code, "policy_id": call.RBAC.PolicyID, "reason": call.RBAC.Reason},
        })
}

// OpenAIClient — клиент OpenAI-совместимого API (/chat/completions).
type OpenAIClient struct {
        baseURL string
        apiKey  string
        model   string
        timeout time.Duration
        http    *http.Client
}

// DefaultLLMBaseURL — базовый URL по умолчанию.
const DefaultLLMBaseURL = "https://api.openai.com/v1"

// NewOpenAIClient создаёт клиента. Возвращает nil, если ключ или модель не заданы
// (тогда агент работает на детерминированном Smart Engine).
func NewOpenAIClient(baseURL, apiKey, model string, timeout time.Duration) *OpenAIClient {
        if strings.TrimSpace(apiKey) == "" || strings.TrimSpace(model) == "" {
                return nil
        }
        if strings.TrimSpace(baseURL) == "" {
                baseURL = DefaultLLMBaseURL
        }
        if timeout <= 0 {
                timeout = 12 * time.Second
        }
        return &OpenAIClient{
                baseURL: strings.TrimRight(strings.TrimSpace(baseURL), "/"), apiKey: strings.TrimSpace(apiKey),
                model: strings.TrimSpace(model), timeout: timeout, http: &http.Client{},
        }
}

// Model возвращает имя модели.
func (c *OpenAIClient) Model() string { return c.model }

// Complete выполняет запрос /chat/completions с таймаутом.
func (c *OpenAIClient) Complete(ctx context.Context, req LLMRequest) (LLMResponse, error) {
        ctx, cancel := context.WithTimeout(ctx, c.timeout)
        defer cancel()

        body := map[string]any{"model": c.model, "messages": req.Messages, "temperature": 0.2}
        if len(req.Tools) > 0 {
                body["tools"] = req.Tools
        }
        raw, err := json.Marshal(body)
        if err != nil {
                return LLMResponse{}, fmt.Errorf("llm: сериализация запроса: %w", err)
        }
        httpReq, err := http.NewRequestWithContext(ctx, http.MethodPost, c.baseURL+"/chat/completions", bytes.NewReader(raw))
        if err != nil {
                return LLMResponse{}, fmt.Errorf("llm: создание запроса: %w", err)
        }
        httpReq.Header.Set("Content-Type", "application/json")
        httpReq.Header.Set("Authorization", "Bearer "+c.apiKey)

        resp, err := c.http.Do(httpReq)
        if err != nil {
                return LLMResponse{}, fmt.Errorf("llm: сеть: %w", err)
        }
        defer resp.Body.Close()
        data, err := io.ReadAll(io.LimitReader(resp.Body, 1<<20))
        if err != nil {
                return LLMResponse{}, fmt.Errorf("llm: чтение ответа: %w", err)
        }
        if resp.StatusCode < 200 || resp.StatusCode > 299 {
                return LLMResponse{}, fmt.Errorf("llm: HTTP %d: %s", resp.StatusCode, truncate(string(data), 200))
        }
        var parsed struct {
                Choices []struct {
                        Message struct {
                                Content   *string       `json:"content"`
                                ToolCalls []LLMToolCall `json:"tool_calls"`
                        } `json:"message"`
                } `json:"choices"`
        }
        if err := json.Unmarshal(data, &parsed); err != nil {
                return LLMResponse{}, fmt.Errorf("llm: разбор ответа: %w", err)
        }
        if len(parsed.Choices) == 0 {
                return LLMResponse{}, errors.New("llm: пустой ответ (нет choices)")
        }
        msg := parsed.Choices[0].Message
        out := LLMResponse{ToolCalls: msg.ToolCalls}
        if msg.Content != nil {
                out.Content = strings.TrimSpace(*msg.Content)
        }
        return out, nil
}

var repoRe = regexp.MustCompile(`[a-z][a-z0-9]*(?:-[a-z0-9]+)+`)

// AbsenceDraft — черновик заявки на отгул, извлечённый из текста.
type AbsenceDraft struct {
        Date      string
        TimeFrom  string
        TimeTo    string
        Reason    string
        DateFound bool
}

var (
        explicitDateRe = regexp.MustCompile(`(?:^|\D)(\d{1,2})[./](\d{1,2})(?:[./](\d{4}))?(?:\D|$)`)
        timeRangeRe    = regexp.MustCompile(`с\s*(\d{1,2})(?::(\d{2}))?\s*(?:до|по)\s*(\d{1,2})(?::(\d{2}))?`)
        weekdays       = []struct {
                stem string
                day  time.Weekday
        }{
                {"понедельник", time.Monday}, {"вторник", time.Tuesday}, {"сред", time.Wednesday},
                {"четверг", time.Thursday}, {"пятниц", time.Friday}, {"суббот", time.Saturday}, {"воскресен", time.Sunday},
        }
        weekdayShort = []string{"вс", "пн", "вт", "ср", "чт", "пт", "сб"}
)

// ParseAbsence извлекает дату, время и причину отгула из свободного текста.
func ParseAbsence(text string, now time.Time) AbsenceDraft {
        normalized, tokens := domain.Normalize(text), domain.Tokens(text)
        d := AbsenceDraft{TimeFrom: "10:00", TimeTo: "19:00", Reason: "Личные обстоятельства"}
        today := time.Date(now.Year(), now.Month(), now.Day(), 0, 0, 0, 0, now.Location())

        setDate := func(t time.Time) {
                d.Date, d.DateFound = t.Format("2006-01-02"), true
        }
        switch m := explicitDateRe.FindStringSubmatch(normalized); {
        case m != nil:
                day, _ := strconv.Atoi(m[1])
                month, _ := strconv.Atoi(m[2])
                year := today.Year()
                if m[3] != "" {
                        year, _ = strconv.Atoi(m[3])
                }
                t := time.Date(year, time.Month(month), day, 0, 0, 0, 0, now.Location())
                if t.Day() == day && int(t.Month()) == month { // отсекаем 31.02 и подобное
                        if m[3] == "" && t.Before(today) {
                                t = t.AddDate(1, 0, 0)
                        }
                        setDate(t)
                }
        case domain.MatchTerm("послезавтра$", normalized, tokens):
                setDate(today.AddDate(0, 0, 2))
        case domain.MatchTerm("завтра$", normalized, tokens):
                setDate(today.AddDate(0, 0, 1))
        case domain.MatchTerm("сегодня$", normalized, tokens):
                setDate(today)
        default:
                for _, w := range weekdays {
                        if domain.MatchTerm(w.stem, normalized, tokens) {
                                ahead := (int(w.day) - int(today.Weekday()) + 7) % 7
                                if ahead == 0 {
                                        ahead = 7
                                }
                                setDate(today.AddDate(0, 0, ahead))
                                break
                        }
                }
        }

        switch m := timeRangeRe.FindStringSubmatch(normalized); {
        case m != nil:
                from, to := clock(m[1], m[2]), clock(m[3], m[4])
                if from != "" && to != "" && to > from {
                        d.TimeFrom, d.TimeTo = from, to
                }
        case strings.Contains(normalized, "после обеда") || strings.Contains(normalized, "второй половин"):
                d.TimeFrom, d.TimeTo = "14:00", "18:00"
        case strings.Contains(normalized, "до обеда") || strings.Contains(normalized, "первой половин") || domain.MatchTerm("утром$", normalized, tokens):
                d.TimeFrom, d.TimeTo = "10:00", "14:00"
        }

        switch {
        case domain.MatchGroup("зачет", normalized, tokens):
                d.Reason = "Зачёт в университете"
        case domain.MatchGroup("экзамен|сесси", normalized, tokens):
                d.Reason = "Экзамен в университете"
        case domain.MatchGroup("врач|поликлиник|стоматолог|больниц", normalized, tokens):
                d.Reason = "Визит к врачу"
        case domain.MatchGroup("универ|учеб|пары$|диплом", normalized, tokens):
                d.Reason = "Учёба в университете"
        }
        return d
}

func clock(h, m string) string {
        hour, err := strconv.Atoi(h)
        if err != nil || hour > 23 {
                return ""
        }
        minute := 0
        if m != "" {
                minute, _ = strconv.Atoi(m)
                if minute > 59 {
                        return ""
                }
        }
        return fmt.Sprintf("%02d:%02d", hour, minute)
}

// HumanDay форматирует дату YYYY-MM-DD как «пт, 13.11».
func HumanDay(date string) string {
        t, err := time.Parse("2006-01-02", date)
        if err != nil {
                return date
        }
        return fmt.Sprintf("%s, %s", weekdayShort[t.Weekday()], t.Format("02.01"))
}

func truncate(s string, max int) string {
        r := []rune(strings.TrimSpace(s))
        if len(r) <= max {
                return string(r)
        }
        return string(r[:max])
}

// retry повторяет подключение с паузой (PostgreSQL/Redis в docker compose стартуют не мгновенно).
func retry[T any](ctx context.Context, attempts int, backoff time.Duration, open func(context.Context) (T, error)) (T, error) {
        var zero T
        var err error
        for i := 0; i < attempts; i++ {
                var v T
                if v, err = open(ctx); err == nil {
                        return v, nil
                }
                if i == attempts-1 {
                        break
                }
                select {
                case <-ctx.Done():
                        return zero, ctx.Err()
                case <-time.After(backoff):
                }
        }
        return zero, err
}

func openStore(ctx context.Context, cfg config.Config, log *slog.Logger) (domain.Store, error) {
        if cfg.DatabaseURL == "" {
                log.Info("хранилище: in-memory (DATABASE_URL не задан)")
                return memory.New(), nil
        }
        store, err := retry(ctx, cfg.ConnectAttempts, cfg.ConnectBackoff, func(ctx context.Context) (*postgres.Store, error) {
                return postgres.Open(ctx, cfg.DatabaseURL)
        })
        if err == nil {
                log.Info("хранилище: PostgreSQL, миграции применены")
                return store, nil
        }
        if cfg.StorageStrict {
                return nil, fmt.Errorf("PostgreSQL недоступен (STORAGE_STRICT=true): %w", err)
        }
        log.Warn("PostgreSQL недоступен — откат на in-memory хранилище", "error", err.Error())
        return memory.New(), nil
}

func openCache(ctx context.Context, cfg config.Config, log *slog.Logger) (domain.Cache, error) {
        if cfg.RedisURL == "" {
                log.Info("кэш: in-memory (REDIS_URL не задан)")
                return cache.NewMemory(), nil
        }
        c, err := retry(ctx, cfg.ConnectAttempts, cfg.ConnectBackoff, func(ctx context.Context) (*cache.Redis, error) {
                return cache.OpenRedis(ctx, cfg.RedisURL)
        })
        if err == nil {
                log.Info("кэш: Redis")
                return c, nil
        }
        if cfg.StorageStrict {
                return nil, fmt.Errorf("Redis недоступен (STORAGE_STRICT=true): %w", err)
        }
        log.Warn("Redis недоступен — откат на in-memory кэш", "error", err.Error())
        return cache.NewMemory(), nil
}

// Build собирает приложение по конфигурации.
func Build(ctx context.Context, cfg config.Config, log *slog.Logger, o Overrides) (*App, error) {
        bundle, err := spec.Load()
        if err != nil {
                return nil, err
        }
        store, err := openStore(ctx, cfg, log)
        if err != nil {
                return nil, err
        }
        kv, err := openCache(ctx, cfg, log)
        if err != nil {
                _ = store.Close()
                return nil, err
        }
        application, err := Assemble(bundle, store, kv, cfg, log, o)
        if err != nil {
                _ = store.Close()
                _ = kv.Close()
                return nil, err
        }
        if err := application.Service.EnsureSeed(ctx); err != nil {
                _ = application.Close()
                return nil, fmt.Errorf("загрузка синтетических данных: %w", err)
        }
        return application, nil
}

// Assemble связывает компоненты поверх готовых хранилища и кэша.
func Assemble(bundle *spec.Bundle, store domain.Store, kv domain.Cache, cfg config.Config, log *slog.Logger, o Overrides) (*App, error) {
        if log == nil {
                log = slog.New(slog.NewTextHandler(io.Discard, nil))
        }
        clock, ids := o.Clock, o.IDs
        if clock == nil {
                clock = domain.SystemClock{}
        }
        if ids == nil {
                ids = domain.RandomIDs{}
        }
        data := seed.Data()

        registry := mcp.NewRegistry(bundle.Tools)
        services := mcp.NewServices(mcp.Deps{Store: store, Cache: kv, Clock: clock, IDs: ids, Seed: data})
        if err := services.BindAll(registry); err != nil {
                return nil, err
        }
        engine := rbac.New(bundle.RBAC)
        telemetry := mcp.NewTelemetry(store, clock, ids)
        gateway := mcp.NewGateway(registry, engine, telemetry, store, clock, ids)

        llm := o.LLM
        if llm == nil {
                if client := agent.NewOpenAIClient(cfg.LLMBaseURL, cfg.LLMAPIKey, cfg.LLMModel, cfg.LLMTimeout); client != nil {
                        llm = client
                        log.Info("LLM подключена", "model", cfg.LLMModel)
                } else {
                        log.Info("LLM не настроена — работает детерминированный Smart Engine")
                }
        }
        buddy := agent.New(agent.Options{
                Spec: &bundle.Agent, Tools: gateway, Catalog: registry, Flags: services.Flags,
                LLM: llm, Clock: clock, Delay: cfg.SmartDelay,
        })
        service := usecase.New(usecase.Deps{
                Spec: bundle, Store: store, Cache: kv, Directory: domain.NewDirectory(data.Employees),
                RBAC: engine, Gateway: gateway, Services: services, Agent: buddy, Events: telemetry,
                Clock: clock, IDs: ids, Seed: data, CronID: seed.CronID,
        })
        handler := httpapi.New(httpapi.Options{
                Service: service, Logger: log, CORSOrigins: cfg.CORSOrigins, StreamChunkDelay: cfg.StreamChunkDelay,
        })
        return &App{Handler: handler, Service: service, Store: store, Cache: kv}, nil
}

// Memory — in-memory кэш с TTL.
type Memory struct {
        mu    sync.Mutex
        items map[string]memItem
        now   func() time.Time
}

type memItem struct {
        value   string
        expires time.Time
}

// NewMemory создаёт in-memory кэш.
func NewMemory() *Memory {
        return &Memory{items: map[string]memItem{}, now: time.Now}
}

// SetClock подменяет источник времени (для тестов TTL).
func (m *Memory) SetClock(now func() time.Time) { m.now = now }

// Get возвращает значение, если оно есть и не истекло.
func (m *Memory) Get(_ context.Context, key string) (string, bool, error) {
        m.mu.Lock()
        defer m.mu.Unlock()
        it, ok := m.items[key]
        if !ok {
                return "", false, nil
        }
        if !it.expires.IsZero() && !m.now().Before(it.expires) {
                delete(m.items, key)
                return "", false, nil
        }
        return it.value, true, nil
}

// Set сохраняет значение; ttl <= 0 — без срока жизни.
func (m *Memory) Set(_ context.Context, key, value string, ttl time.Duration) error {
        m.mu.Lock()
        defer m.mu.Unlock()
        it := memItem{value: value}
        if ttl > 0 {
                it.expires = m.now().Add(ttl)
        }
        m.items[key] = it
        return nil
}

// Del удаляет ключи.
func (m *Memory) Del(_ context.Context, keys ...string) error {
        m.mu.Lock()
        defer m.mu.Unlock()
        for _, k := range keys {
                delete(m.items, k)
        }
        return nil
}

// Ping всегда успешен.
func (m *Memory) Ping(context.Context) error { return nil }

// Name возвращает имя реализации.
func (m *Memory) Name() string { return "memory" }

// Close ничего не делает.
func (m *Memory) Close() error { return nil }

// Redis — кэш на Redis.
type Redis struct {
        client *redis.Client
}

// OpenRedis подключается к Redis по URL (redis://host:port/db) и проверяет соединение.
func OpenRedis(ctx context.Context, url string) (*Redis, error) {
        opts, err := redis.ParseURL(url)
        if err != nil {
                return nil, fmt.Errorf("redis: разбор URL: %w", err)
        }
        opts.DialTimeout = 2 * time.Second
        opts.ReadTimeout = time.Second
        opts.WriteTimeout = time.Second
        r := &Redis{client: redis.NewClient(opts)}
        if err := r.Ping(ctx); err != nil {
                _ = r.client.Close()
                return nil, fmt.Errorf("redis: недоступен: %w", err)
        }
        return r, nil
}

// Get возвращает значение ключа.
func (r *Redis) Get(ctx context.Context, key string) (string, bool, error) {
        v, err := r.client.Get(ctx, key).Result()
        if errors.Is(err, redis.Nil) {
                return "", false, nil
        }
        if err != nil {
                return "", false, fmt.Errorf("redis: get %s: %w", key, err)
        }
        return v, true, nil
}

// Set сохраняет значение с TTL (0 — бессрочно).
func (r *Redis) Set(ctx context.Context, key, value string, ttl time.Duration) error {
        if ttl < 0 {
                ttl = 0
        }
        if err := r.client.Set(ctx, key, value, ttl).Err(); err != nil {
                return fmt.Errorf("redis: set %s: %w", key, err)
        }
        return nil
}

// Del удаляет ключи.
func (r *Redis) Del(ctx context.Context, keys ...string) error {
        if len(keys) == 0 {
                return nil
        }
        if err := r.client.Del(ctx, keys...).Err(); err != nil {
                return fmt.Errorf("redis: del: %w", err)
        }
        return nil
}

// Ping проверяет соединение.
func (r *Redis) Ping(ctx context.Context) error { return r.client.Ping(ctx).Err() }

// Name возвращает имя реализации.
func (r *Redis) Name() string { return "redis" }

// Close закрывает клиент.
func (r *Redis) Close() error { return r.client.Close() }

// Getenv — функция чтения переменной окружения (os.Getenv в бою).
type Getenv func(string) string

func intEnv(get Getenv, key string, def int) int {
        if v, err := strconv.Atoi(strings.TrimSpace(get(key))); err == nil {
                return v
        }
        return def
}

func boolEnv(get Getenv, key string) bool {
        switch strings.ToLower(strings.TrimSpace(get(key))) {
        case "1", "true", "yes", "on":
                return true
        }
        return false
}

func strEnv(get Getenv, key, def string) string {
        if v := strings.TrimSpace(get(key)); v != "" {
                return v
        }
        return def
}

// Load собирает конфигурацию из окружения.
func Load(get Getenv) Config {
        c := Config{
                Port:             strEnv(get, "PORT", "8080"),
                DatabaseURL:      strings.TrimSpace(get("DATABASE_URL")),
                RedisURL:         strings.TrimSpace(get("REDIS_URL")),
                StorageStrict:    boolEnv(get, "STORAGE_STRICT"),
                ConnectAttempts:  intEnv(get, "STORAGE_CONNECT_ATTEMPTS", 10),
                ConnectBackoff:   time.Duration(intEnv(get, "STORAGE_CONNECT_BACKOFF_MS", 1000)) * time.Millisecond,
                LLMAPIKey:        strings.TrimSpace(get("LLM_API_KEY")),
                LLMBaseURL:       strings.TrimSpace(get("LLM_BASE_URL")),
                LLMModel:         strings.TrimSpace(get("LLM_MODEL")),
                LLMTimeout:       time.Duration(intEnv(get, "LLM_TIMEOUT_MS", 12000)) * time.Millisecond,
                SmartDelay:       time.Duration(intEnv(get, "SMART_DELAY_MS", -1)) * time.Millisecond,
                StreamChunkDelay: time.Duration(intEnv(get, "STREAM_CHUNK_DELAY_MS", 12)) * time.Millisecond,
        }
        for _, o := range strings.Split(strEnv(get, "CORS_ORIGINS", "*"), ",") {
                if o = strings.TrimSpace(o); o != "" {
                        c.CORSOrigins = append(c.CORSOrigins, o)
                }
        }
        if c.ConnectAttempts < 1 {
                c.ConnectAttempts = 1
        }
        return c
}

// ChecklistStatus — статус шага чек-листа адаптации.
type ChecklistStatus string

// Статусы шага чек-листа.
const (
        StatusPending    ChecklistStatus = "pending"
        StatusInProgress ChecklistStatus = "in_progress"
        StatusCompleted  ChecklistStatus = "completed"
)

// Valid сообщает, известен ли статус.
func (s ChecklistStatus) Valid() bool {
        return s == StatusPending || s == StatusInProgress || s == StatusCompleted
}

func (s ChecklistStatus) rank() int {
        switch s {
        case StatusInProgress:
                return 1
        case StatusCompleted:
                return 2
        default:
                return 0
        }
}

// ChecklistItem — шаг чек-листа адаптации стажёра.
type ChecklistItem struct {
        ID          string          `json:"id"`
        Position    int             `json:"position"`
        Week        int             `json:"week"`
        Title       string          `json:"title"`
        Description string          `json:"description"`
        Status      ChecklistStatus `json:"status"`
        Scenario    string          `json:"scenario,omitempty"`
        Evidence    string          `json:"evidence,omitempty"`
        UpdatedAt   time.Time       `json:"updated_at"`
        CompletedAt *time.Time      `json:"completed_at,omitempty"`
}

// ErrInvalidTransition — попытка недопустимого перехода статуса.
var ErrInvalidTransition = errors.New("недопустимый переход статуса чек-листа")

// Advance переводит шаг в новый статус. Статус может только расти
// (pending → in_progress → completed); повтор текущего статуса — no-op.
func (c ChecklistItem) Advance(to ChecklistStatus, evidence string, now time.Time) (ChecklistItem, bool, error) {
        if !to.Valid() {
                return c, false, fmt.Errorf("%w: неизвестный статус %q", ErrInvalidTransition, to)
        }
        if to.rank() < c.Status.rank() {
                return c, false, fmt.Errorf("%w: %s → %s", ErrInvalidTransition, c.Status, to)
        }
        if to == c.Status {
                return c, false, nil
        }
        c.Status = to
        c.UpdatedAt = now
        if evidence != "" {
                c.Evidence = evidence
        }
        if to == StatusCompleted {
                t := now
                c.CompletedAt = &t
        }
        return c, true, nil
}

// Progress — агрегированный прогресс чек-листа.
type Progress struct {
        Completed  int `json:"completed"`
        InProgress int `json:"in_progress"`
        Total      int `json:"total"`
        Percent    int `json:"percent"`
}

// ComputeProgress считает прогресс по шагам чек-листа.
func ComputeProgress(items []ChecklistItem) Progress {
        p := Progress{Total: len(items)}
        for _, it := range items {
                switch it.Status {
                case StatusCompleted:
                        p.Completed++
                case StatusInProgress:
                        p.InProgress++
                }
        }
        if p.Total > 0 {
                p.Percent = p.Completed * 100 / p.Total
        }
        return p
}

// NextStep возвращает первый невыполненный шаг (по позиции) или nil.
func NextStep(items []ChecklistItem) *ChecklistItem {
        var next *ChecklistItem
        for i := range items {
                if items[i].Status == StatusCompleted {
                        continue
                }
                if next == nil || items[i].Position < next.Position {
                        next = &items[i]
                }
        }
        return next
}

        dateRe = regexp.MustCompile(`^\d{4}-\d{2}-\d{2}$`)
        timeRe = regexp.MustCompile(`^([01]\d|2[0-3]):[0-5]\d$`)
)

// ErrValidation — ошибка валидации доменной сущности.
var ErrValidation = errors.New("ошибка валидации")

// Validate проверяет заявку: формат даты и времени, дата не в прошлом, интервал корректен.
func (a AbsenceRequest) Validate(now time.Time) error {
        if strings.TrimSpace(a.InternID) == "" {
                return fmt.Errorf("%w: не указан сотрудник", ErrValidation)
        }
        if !dateRe.MatchString(a.Date) {
                return fmt.Errorf("%w: дата должна быть в формате ГГГГ-ММ-ДД", ErrValidation)
        }
        day, err := time.ParseInLocation("2006-01-02", a.Date, now.Location())
        if err != nil {
                return fmt.Errorf("%w: некорректная дата %q", ErrValidation, a.Date)
        }
        today := time.Date(now.Year(), now.Month(), now.Day(), 0, 0, 0, 0, now.Location())
        if day.Before(today) {
                return fmt.Errorf("%w: дата отгула уже прошла", ErrValidation)
        }
        if (a.TimeFrom == "") != (a.TimeTo == "") {
                return fmt.Errorf("%w: укажите и начало, и окончание отгула", ErrValidation)
        }
        if a.TimeFrom != "" {
                if !timeRe.MatchString(a.TimeFrom) || !timeRe.MatchString(a.TimeTo) {
                        return fmt.Errorf("%w: время должно быть в формате ЧЧ:ММ", ErrValidation)
                }
                if a.TimeTo <= a.TimeFrom {
                        return fmt.Errorf("%w: окончание должно быть позже начала", ErrValidation)
                }
        }
        if n := len([]rune(strings.TrimSpace(a.Reason))); n < 3 || n > 300 {
                return fmt.Errorf("%w: причина должна быть от 3 до 300 символов", ErrValidation)
        }
        return nil
}

// MaskToken маскирует токен, оставляя префикс и 4 последних символа.
func MaskToken(token string) string {
        const keep = 4
        r := []rune(token)
        if len(r) <= keep {
                return strings.Repeat("•", len(r))
        }
        prefix := ""
        if i := strings.LastIndex(token, "-"); i > 0 && i < len(token)-keep {
                prefix = token[:i+1]
        }
        return prefix + "••••" + string(r[len(r)-keep:])
}

// ComputeMetrics считает метрики по событиям телеметрии.
// Событие учитывается в экономии, если в payload есть saving_key из спецификации.
func ComputeMetrics(in MetricsInput) MetricsSummary {
        m := MetricsSummary{
                DocsRead: in.DocsRead, DocsTotal: in.DocsTotal,
                ChecklistPercent:       in.Progress.Percent,
                MentorHoursBaselineMin: in.BaselineHoursMin,
                MentorHoursBaselineMax: in.BaselineHoursMax,
                UnansweredQuestions:    in.OpenQuestions,
        }
        var latencySum float64
        var latencyN int
        for _, e := range in.Events {
                if e.Kind == EventKindMCP {
                        m.ToolCalls++
                }
                if v, ok := e.Payload["latency_ms"].(float64); ok && e.Kind == EventKindAgent {
                        latencySum += v
                        latencyN++
                }
                key, _ := e.Payload["saving_key"].(string)
                if key == "" {
                        continue
                }
                m.MentorMinutesSaved += in.MinutesSaved[key]
                switch key {
                case SavingQuestion:
                        m.QuestionsAnswered++
                case SavingAccess, SavingAbsence:
                        m.ActionsAutomated++
                case SavingBrief:
                        m.BriefsPrepared++
                case SavingBlocked:
                        m.RBACBlocks++
                }
        }
        if latencyN > 0 {
                m.AvgResponseMS = round1(latencySum / float64(latencyN))
        }
        if avg := float64(in.BaselineHoursMin+in.BaselineHoursMax) / 2 * 60; avg > 0 {
                m.SavingPercent = round1(float64(m.MentorMinutesSaved) / avg * 100)
        }
        return m
}

func round1(v float64) float64 {
        return float64(int(v*10+0.5)) / 10
}

// NewRecord сериализует значение в запись хранилища.
func NewRecord(kind Kind, id, ownerID string, at time.Time, v any) (Record, error) {
        if id == "" {
                return Record{}, fmt.Errorf("%w: пустой идентификатор записи %s", ErrValidation, kind)
        }
        raw, err := json.Marshal(v)
        if err != nil {
                return Record{}, fmt.Errorf("сериализация %s/%s: %w", kind, id, err)
        }
        return Record{Kind: kind, ID: id, OwnerID: ownerID, CreatedAt: at, Payload: raw}, nil
}

// Save сохраняет типизированное значение.
func Save(ctx context.Context, s Store, kind Kind, id, ownerID string, at time.Time, v any) error {
        rec, err := NewRecord(kind, id, ownerID, at, v)
        if err != nil {
                return err
        }
        return s.Put(ctx, rec)
}

// Load читает типизированные значения владельца.
func Load[T any](ctx context.Context, s Store, kind Kind, ownerID string) ([]T, error) {
        recs, err := s.List(ctx, kind, ownerID)
        if err != nil {
                return nil, err
        }
        out := make([]T, 0, len(recs))
        for _, r := range recs {
                var v T
                if err := json.Unmarshal(r.Payload, &v); err != nil {
                        return nil, fmt.Errorf("десериализация %s/%s: %w", kind, r.ID, err)
                }
                out = append(out, v)
        }
        return out, nil
}

// LoadOne читает одно типизированное значение по идентификатору.

                return v, err
        }
        if err := json.Unmarshal(rec.Payload, &v); err != nil {
                return v, fmt.Errorf("десериализация %s/%s: %w", kind, id, err)
        }
        return v, nil
}

// Normalize приводит текст к нижнему регистру, заменяет «ё» на «е» и схлопывает пробелы.
func Normalize(s string) string {
        s = strings.ToLower(s)
        s = strings.ReplaceAll(s, "ё", "е")
        return strings.Join(strings.Fields(s), " ")
}

// Tokens разбивает нормализованный текст на слова (буквы и цифры).
func Tokens(s string) []string {
        return strings.FieldsFunc(Normalize(s), func(r rune) bool {
                return !unicode.IsLetter(r) && !unicode.IsDigit(r)
        })
}

// MatchTerm проверяет один термин триггера:
//   - термин с пробелом или дефисом — подстрока нормализованного текста;
//   - «слово$» — точное совпадение слова;
//   - иначе — префикс любого слова.
func MatchTerm(term, normalized string, tokens []string) bool {
        term = Normalize(term)
        if term == "" {
                return false
        }
        if strings.ContainsAny(term, " -") {
                return strings.Contains(normalized, term)
        }
        exact := strings.HasSuffix(term, "$")
        term = strings.TrimSuffix(term, "$")
        for _, t := range tokens {
                if exact && t == term {
                        return true
                }
                if !exact && strings.HasPrefix(t, term) {
                        return true
                }
        }
        return false
}

// MatchGroup проверяет группу альтернатив «a|b|c»: достаточно одного совпадения.
func MatchGroup(group, normalized string, tokens []string) bool {
        for _, term := range strings.Split(group, "|") {
                if MatchTerm(term, normalized, tokens) {
                        return true
                }
        }
        return false
}

// MatchTrigger проверяет триггер: должны совпасть все группы.
func MatchTrigger(trigger []string, text string) bool {
        if len(trigger) == 0 {
                return false
        }
        normalized := Normalize(text)
        tokens := Tokens(text)
        for _, group := range trigger {
                if !MatchGroup(group, normalized, tokens) {
                        return false
                }
        }
        return true
}

// Options — параметры HTTP-слоя.
type Options struct {
        Service *usecase.Service
        Logger  *slog.Logger
        // CORSOrigins — разрешённые источники; "*" — любой (демо-режим).
        CORSOrigins []string
        // StreamChunkDelay — пауза между фрагментами текста в SSE (эффект «печатает»).
        StreamChunkDelay time.Duration
}

// Server — HTTP-обработчик API.
type Server struct {
        svc        *usecase.Service
        log        *slog.Logger
        origins    []string
        chunkDelay time.Duration
        mux        *http.ServeMux
}

const maxBodyBytes = 64 << 10

type ctxKey struct{}

// New создаёт HTTP-обработчик со всеми маршрутами.
func New(o Options) http.Handler {
        if o.Logger == nil {
                o.Logger = slog.New(slog.NewTextHandler(io.Discard, nil))
        }
        if len(o.CORSOrigins) == 0 {
                o.CORSOrigins = []string{"*"}
        }
        s := &Server{svc: o.Service, log: o.Logger, origins: o.CORSOrigins, chunkDelay: o.StreamChunkDelay, mux: http.NewServeMux()}

        s.mux.HandleFunc("GET /healthz", s.health)
        s.mux.Handle("GET /api/v1/workspace", s.auth(s.workspace))
        s.mux.Handle("POST /api/v1/chat", s.auth(s.chat))
        s.mux.Handle("POST /api/v1/chat/stream", s.auth(s.chatStream))
        s.mux.Handle("POST /api/v1/mcp/execute", s.auth(s.execute))
        s.mux.Handle("POST /api/v1/absences/review", s.auth(s.reviewAbsence))
        s.mux.Handle("POST /api/v1/cron/trigger", s.auth(s.cron))
        s.mux.Handle("POST /api/v1/reset", s.auth(s.reset))
        s.mux.Handle("GET /api/v1/telemetry", s.auth(s.telemetry))
        s.mux.Handle("PUT /api/v1/feature-flags/{key}", s.auth(s.setFlag))
        s.mux.Handle("POST /api/v1/events", s.auth(s.trackEvent))
        s.mux.Handle("POST /mcp", s.auth(s.mcpRPC))
        s.mux.HandleFunc("/", func(w http.ResponseWriter, _ *http.Request) {
                writeError(w, http.StatusNotFound, "NOT_FOUND", "Маршрут не найден", nil)
        })
        return s.middleware(s.mux)
}

// statusRecorder запоминает код ответа для журнала.
type statusRecorder struct {
        http.ResponseWriter
        status int
}

func (r *statusRecorder) WriteHeader(code int) {
        r.status = code
        r.ResponseWriter.WriteHeader(code)
}

// Flush пробрасывает сброс буфера (нужно для SSE).
func (r *statusRecorder) Flush() {
        if f, ok := r.ResponseWriter.(http.Flusher); ok {
                f.Flush()
        }
}

func (s *Server) allowOrigin(origin string) string {
        for _, o := range s.origins {
                if o == "*" {
                        return "*"
                }
                if origin != "" && strings.EqualFold(o, origin) {
                        return origin
                }
        }
        return ""
}

// middleware: восстановление после паники, CORS, лимит тела, журнал и время ответа.
func (s *Server) middleware(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                started := time.Now()
                rec := &statusRecorder{ResponseWriter: w, status: http.StatusOK}
                defer func() {
                        if p := recover(); p != nil {
                                s.log.Error("panic", "path", r.URL.Path, "panic", fmt.Sprint(p))
                                writeError(rec, http.StatusInternalServerError, "INTERNAL", "Внутренняя ошибка сервера", nil)
                        }
                        s.log.Info("http", "method", r.Method, "path", r.URL.Path, "status", rec.status,
                                "ms", float64(time.Since(started).Microseconds())/1000)
                }()

                if allowed := s.allowOrigin(r.Header.Get("Origin")); allowed != "" {
                        rec.Header().Set("Access-Control-Allow-Origin", allowed)
                        rec.Header().Set("Vary", "Origin")
                        rec.Header().Set("Access-Control-Allow-Headers", "Authorization, Content-Type")
                        rec.Header().Set("Access-Control-Allow-Methods", "GET, POST, PUT, OPTIONS")
                        rec.Header().Set("Access-Control-Max-Age", "600")
                }
                rec.Header().Set("X-Content-Type-Options", "nosniff")
                if r.Method == http.MethodOptions {
                        rec.WriteHeader(http.StatusNoContent)
                        return
                }
                r.Body = http.MaxBytesReader(rec, r.Body, maxBodyBytes)
                next.ServeHTTP(rec, r)
        })
}

// auth — имитация корпоративной авторизации: демо-токен → сотрудник → роль.
func (s *Server) auth(next func(http.ResponseWriter, *http.Request, domain.Principal)) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                token := strings.TrimSpace(strings.TrimPrefix(r.Header.Get("Authorization"), "Bearer "))
                p, err := s.svc.Authenticate(token)
                if err != nil {
                        s.fail(w, r, err)
                        return
                }
                next(w, r.WithContext(context.WithValue(r.Context(), ctxKey{}, p)), p)
        })
}

type errorBody struct {
        Code    string              `json:"code"`
        Message string              `json:"message"`
        RBAC    *domain.RBACVerdict `json:"rbac,omitempty"`
}

type errorEnvelope struct {
        Error  errorBody `json:"error"`
        Result any       `json:"result,omitempty"`
}

func writeJSON(w http.ResponseWriter, status int, v any) {
        w.Header().Set("Content-Type", "application/json; charset=utf-8")
        w.WriteHeader(status)
        _ = json.NewEncoder(w).Encode(v)
}

func writeError(w http.ResponseWriter, status int, code, message string, verdict *domain.RBACVerdict) {
        writeJSON(w, status, errorEnvelope{Error: errorBody{Code: code, Message: message, RBAC: verdict}})
}

// classify сопоставляет ошибку прикладного слоя HTTP-статусу и коду.
func classify(err error) (int, errorBody) {
        var forbidden *usecase.ForbiddenError
        var tooLarge *http.MaxBytesError
        switch {
        case errors.Is(err, usecase.ErrUnauthorized):
                return http.StatusUnauthorized, errorBody{Code: "UNAUTHORIZED", Message: err.Error()}
        case errors.As(err, &forbidden):
                v := forbidden.Verdict
                return http.StatusForbidden, errorBody{Code: "FORBIDDEN", Message: v.Reason, RBAC: &v}
        case errors.Is(err, domain.ErrValidation):
                return http.StatusUnprocessableEntity, errorBody{Code: "VALIDATION_ERROR", Message: err.Error()}
        case errors.Is(err, domain.ErrNotFound):
                return http.StatusNotFound, errorBody{Code: "NOT_FOUND", Message: err.Error()}
        case errors.As(err, &tooLarge):
                return http.StatusRequestEntityTooLarge, errorBody{Code: "PAYLOAD_TOO_LARGE", Message: "Слишком большой запрос"}
        default:
                return http.StatusInternalServerError, errorBody{Code: "INTERNAL", Message: "Внутренняя ошибка сервера. Попробуйте ещё раз."}
        }
}

func (s *Server) fail(w http.ResponseWriter, r *http.Request, err error) {
        status, body := classify(err)
        if status >= 500 {
                s.log.Error("request failed", "path", r.URL.Path, "error", err.Error())
        }
        writeJSON(w, status, errorEnvelope{Error: body})
}

// decode читает JSON-тело запроса в структуру.
func decode(r *http.Request, dst any) error {
        dec := json.NewDecoder(r.Body)
        if err := dec.Decode(dst); err != nil {
                var tooLarge *http.MaxBytesError
                if errors.As(err, &tooLarge) {
                        return err
                }
                return fmt.Errorf("%w: некорректный JSON в теле запроса", domain.ErrValidation)
        }
        return nil
}

func (s *Server) health(w http.ResponseWriter, r *http.Request) {
        info, err := s.svc.Health(r.Context())
        status, state := http.StatusOK, "ok"
        if err != nil {
                status, state = http.StatusServiceUnavailable, "degraded"
        }
        writeJSON(w, status, map[string]any{"status": state, "store": info.Store, "cache": info.Cache})
}

func getInternID(r *http.Request, explicit ...string) string {
        for _, id := range explicit {
                if s := strings.TrimSpace(id); s != "" {
                        return s
                }
        }
        if id := strings.TrimSpace(r.URL.Query().Get("intern_id")); id != "" {
                return id
        }
        if id := strings.TrimSpace(r.Header.Get("X-Intern-ID")); id != "" {
                return id
        }
        return ""
}

}

func sseEvent(w http.ResponseWriter, f http.Flusher, event string, data any) {
        raw, _ := json.Marshal(data)
        fmt.Fprintf(w, "event: %s\ndata: %s\n\n", event, raw)
        f.Flush()
}

// chunks делит текст на фрагменты по границам слов для потоковой отдачи.
func chunks(text string, size int) []string {
        var out []string
        var b strings.Builder
        n := 0
        for _, part := range strings.SplitAfter(text, " ") {
                b.WriteString(part)
                n += len([]rune(part))
                if n >= size {
                        out = append(out, b.String())
                        b.Reset()
                        n = 0
                }
        }
        if b.Len() > 0 {
                out = append(out, b.String())
        }
        return out
}

// chatStream — потоковый вариант чата (SSE): этапы обработки (stage), фрагменты ответа (delta)
// и итог (done). Интерфейс не блокируется и показывает статус ожидания.
func (s *Server) chatStream(w http.ResponseWriter, r *http.Request, p domain.Principal) {
        var req chatRequest
        if err := decode(r, &req); err != nil {
                s.fail(w, r, err)
                return
        }
        flusher, ok := w.(http.Flusher)
        if !ok {
                s.fail(w, r, errors.New("потоковая передача не поддерживается"))
                return
        }
        w.Header().Set("Content-Type", "text/event-stream; charset=utf-8")
        w.Header().Set("Cache-Control", "no-cache")
        w.Header().Set("X-Accel-Buffering", "no")
        w.WriteHeader(http.StatusOK)
        flusher.Flush()

        internID := getInternID(r, req.InternID)
        res, err := s.svc.Chat(r.Context(), p, req.Message, func(st agent.Stage) { sseEvent(w, flusher, "stage", st) }, internID)
        if err != nil {
                status, body := classify(err)
                if status >= 500 {
                        s.log.Error("chat stream failed", "error", err.Error())
                }
                sseEvent(w, flusher, "error", map[string]any{"status": status, "error": body})
                return
        }
        for _, part := range chunks(res.Message.Text, 24) {
                if r.Context().Err() != nil {
                        return
                }
                sseEvent(w, flusher, "delta", map[string]string{"text": part})
                if s.chunkDelay > 0 {
                        time.Sleep(s.chunkDelay)
                }
        }
        sseEvent(w, flusher, "done", res)
}

// execute — прямое подтверждение интерактивной карточки: вызов MCP-инструмента через RBAC-шлюз.
func (s *Server) execute(w http.ResponseWriter, r *http.Request, p domain.Principal) {
        var req executeRequest
        if err := decode(r, &req); err != nil {
                s.fail(w, r, err)
                return
        }
        internID := getInternID(r, req.InternID)
        res, err := s.svc.Execute(r.Context(), p, req.Tool, req.Arguments, req.MessageID, internID)
        if err != nil {
                s.fail(w, r, err)
                return
        }
        call := res.ToolCalls[0]
        if call.OK() {
                writeJSON(w, http.StatusOK, res)
                return
        }
        status, body := http.StatusBadGateway, errorBody{Code: call.ErrorCode, Message: call.Error}
        switch {
        case call.Status == domain.ToolForbidden:
                status, body.RBAC = http.StatusForbidden, res.RBAC
        case call.Status == domain.ToolInvalid:
                status = http.StatusUnprocessableEntity
        case call.ErrorCode == mcp.CodeDuplicate:
                status = http.StatusConflict
        case call.ErrorCode == mcp.CodeUnknownTool:
                status = http.StatusNotFound
        }
        writeJSON(w, status, errorEnvelope{Error: body, Result: res})
}

// Call — контекст вызова инструмента.
type Call struct {
        Principal domain.Principal
        InternID  string
        Args      map[string]any
        Now       time.Time
}

// Handler — обработчик инструмента.
type Handler func(ctx context.Context, c Call) (map[string]any, error)

// AttrResolver извлекает атрибуты ресурса для ABAC-ограничений RBAC.
type AttrResolver func(args map[string]any) rbac.Resource

// Error — бизнес-ошибка инструмента с кодом из спецификации.
type Error struct {
        Code    string
        Message string
}

func (e *Error) Error() string { return e.Code + ": " + e.Message }

// Errorf создаёт бизнес-ошибку инструмента.
func Errorf(code, format string, args ...any) *Error {
        return &Error{Code: code, Message: fmt.Sprintf(format, args...)}
}

// Коды ошибок шлюза.
const (
        CodeUnknownTool = "UNKNOWN_TOOL"
        CodeForbidden   = "FORBIDDEN"
        CodeInvalidArgs = "INVALID_ARGUMENTS"
        CodeInternal    = "INTERNAL"
        CodeDuplicate   = "DUPLICATE"
        CodeNotFound    = "NOT_FOUND"
)

type boundTool struct {
        spec    spec.Tool
        handler Handler
        attrs   AttrResolver
}

// Registry — реестр MCP-инструментов, построенный по спецификации.
type Registry struct {
        servers []spec.Server
        tools   map[string]*boundTool
        order   []string
}

// NewRegistry создаёт реестр из спецификации (обработчики привязываются через Bind).
func NewRegistry(ts spec.ToolsSpec) *Registry {
        r := &Registry{servers: append([]spec.Server{}, ts.Servers...), tools: map[string]*boundTool{}}
        for _, t := range ts.Tools {
                r.tools[t.Name] = &boundTool{spec: t}
                r.order = append(r.order, t.Name)
        }
        return r
}

// Bind привязывает обработчик к инструменту из спецификации.
func (r *Registry) Bind(name string, h Handler, attrs AttrResolver) error {
        t, ok := r.tools[name]
        if !ok {
                return fmt.Errorf("mcp: инструмент %q отсутствует в спецификации", name)
        }
        if h == nil {
                return fmt.Errorf("mcp: пустой обработчик для %q", name)
        }
        if t.handler != nil {
                return fmt.Errorf("mcp: обработчик для %q уже привязан", name)
        }
        t.handler, t.attrs = h, attrs
        return nil
}

// Unbound возвращает инструменты спецификации без обработчика (должно быть пусто).
func (r *Registry) Unbound() []string {
        var out []string
        for _, name := range r.order {
                if r.tools[name].handler == nil {
                        out = append(out, name)
                }
        }
        sort.Strings(out)
        return out
}

// Tools возвращает контракты инструментов в порядке спецификации.
func (r *Registry) Tools() []spec.Tool {
        out := make([]spec.Tool, 0, len(r.order))
        for _, name := range r.order {
                out = append(out, r.tools[name].spec)
        }
        return out
}

// Tool возвращает контракт инструмента.
func (r *Registry) Tool(name string) (spec.Tool, bool) {
        t, ok := r.tools[name]
        if !ok {
                return spec.Tool{}, false
        }
        return t.spec, true
}

// Servers возвращает описания MCP-серверов.
func (r *Registry) Servers() []spec.Server { return append([]spec.Server{}, r.servers...) }

}

// NewTelemetry создаёт приёмник событий.
func NewTelemetry(store domain.Store, clock domain.Clock, ids domain.IDGenerator) *Telemetry {
        return &Telemetry{store: store, clock: clock, ids: ids}
}

// Track записывает событие. Запись best-effort: сбой телеметрии не ломает сценарий.
func (t *Telemetry) Track(ctx context.Context, internID string, actor domain.Principal, kind, name, severity string, payload map[string]any) domain.TelemetryEvent {
        if payload == nil {
                payload = map[string]any{}
        }
        ev := domain.TelemetryEvent{
                ID: t.ids.NewID("evt"), InternID: internID, ActorID: actor.ID, ActorRole: actor.Role,
                Kind: kind, Name: name, Severity: severity, Payload: payload, CreatedAt: t.clock.Now(),
        }
        _ = domain.Save(ctx, t.store, domain.KindEvent, ev.ID, internID, ev.CreatedAt, ev)
        return ev
}

// Execute выполняет вызов инструмента от имени субъекта. Всегда возвращает запись аудита;
// итог — в полях Status / RBAC / Error.
func (g *Gateway) Execute(ctx context.Context, p domain.Principal, internID, tool string, args map[string]any) domain.MCPToolCall {
        started := time.Now()
        if args == nil {
                args = map[string]any{}
        }
        call := domain.MCPToolCall{
                ID: g.ids.NewID("call"), InternID: internID, ActorID: p.ID, ActorRole: p.Role,
                Tool: tool, Arguments: args, Result: map[string]any{}, CreatedAt: g.clock.Now(),
        }
        if i := strings.Index(tool, "."); i > 0 {
                call.Server = tool[:i]
        }

        bt, known := g.reg.tools[tool]
        switch {
        case !known || bt.handler == nil:
                // Неизвестный инструмент: default deny, обработчика нет.
                call.RBAC = g.rbac.Check(p, tool, rbac.Resource{})
                call.Status, call.ErrorCode = domain.ToolError, CodeUnknownTool
                call.Error = fmt.Sprintf("Инструмент %q не зарегистрирован", tool)
        default:
                call.Server = bt.spec.Server
                attrs := defaultAttrs
                if bt.attrs != nil {
                        attrs = bt.attrs
                }
                call.RBAC = g.rbac.Check(p, tool, attrs(args))
                switch {
                case !call.RBAC.Allowed:
                        call.Status, call.ErrorCode, call.Error = domain.ToolForbidden, CodeForbidden, call.RBAC.Reason
                default:
                        if problems := ValidateArgs(bt.spec.InputSchema, args); len(problems) > 0 {
                                call.Status, call.ErrorCode = domain.ToolInvalid, CodeInvalidArgs
                                call.Error = strings.Join(problems, "; ")
                                break
                        }
                        result, err := bt.handler(ctx, Call{Principal: p, InternID: internID, Args: args, Now: call.CreatedAt})
                        var toolErr *Error
                        switch {
                        case err == nil:
                                call.Status = domain.ToolSuccess
                                if result != nil {
                                        call.Result = result
                                }
                        case errors.As(err, &toolErr):
                                call.Status, call.ErrorCode, call.Error = domain.ToolError, toolErr.Code, toolErr.Message
                                if toolErr.Code == CodeInvalidArgs {
                                        call.Status = domain.ToolInvalid
                                }
                        case errors.Is(err, domain.ErrValidation):
                                call.Status, call.ErrorCode, call.Error = domain.ToolInvalid, CodeInvalidArgs, err.Error()
                        default:
                                call.Status, call.ErrorCode, call.Error = domain.ToolError, CodeInternal, err.Error()
                        }
                }
        }

        call.DurationMS = float64(time.Since(started).Microseconds()) / 1000
        g.audit(ctx, p, internID, &call)
        return call
}

func (g *Gateway) audit(ctx context.Context, p domain.Principal, internID string, call *domain.MCPToolCall) {
        if err := domain.Save(ctx, g.store, domain.KindToolCall, call.ID, internID, call.CreatedAt, call); err != nil {
                call.Error = strings.TrimSpace(call.Error + " (аудит не записан: " + err.Error() + ")")
        }
        v := call.RBAC
        rbacSeverity := domain.SeverityInfo
        if !v.Allowed {
                rbacSeverity = domain.SeverityWarn
        }
        g.tel.Track(ctx, internID, p, domain.EventKindRBAC, "rbac.decision", rbacSeverity, map[string]any{
                "operation": v.Operation, "role": string(v.Role), "allowed": v.Allowed, "code": v.Code,
                "policy_id": v.PolicyID, "reason": v.Reason, "constraint": v.Constraint, "tool_call_id": call.ID,
        })
        if v.Incident {
                g.tel.Track(ctx, internID, p, domain.EventKindRBAC, "rbac.incident", domain.SeveritySecurity, map[string]any{
                        "operation": v.Operation, "role": string(v.Role), "policy_id": v.PolicyID,
                        "reason": v.Reason, "actor": p.Name, "tool_call_id": call.ID, "arguments": call.Arguments,
                })
        }
        severity := domain.SeverityInfo
        if call.Status != domain.ToolSuccess {
                severity = domain.SeverityWarn
        }
        g.tel.Track(ctx, internID, p, domain.EventKindMCP, "mcp.call", severity, map[string]any{
                "tool": call.Tool, "server": call.Server, "status": string(call.Status),
                "error_code": call.ErrorCode, "duration_ms": call.DurationMS, "tool_call_id": call.ID,
        })
}

// Минимальная реализация протокола MCP (JSON-RPC 2.0) поверх шлюза:
// initialize, ping, tools/list, tools/call. Этого достаточно, чтобы подключить
// реестр к корпоративной платформе ИИ-агентов или любому MCP-клиенту (POST /mcp).

// ProtocolVersion — версия протокола MCP, объявляемая сервером.
const ProtocolVersion = "2025-06-18"

// Коды ошибок JSON-RPC.
const (
        rpcParseError     = -32700
        rpcInvalidRequest = -32600
        rpcMethodNotFound = -32601
        rpcInvalidParams  = -32602
)

// RPCRequest — запрос JSON-RPC 2.0.
type RPCRequest struct {
        JSONRPC string          `json:"jsonrpc"`
        ID      json.RawMessage `json:"id,omitempty"`
        Method  string          `json:"method"`
        Params  json.RawMessage `json:"params,omitempty"`
}

// RPCError — ошибка JSON-RPC 2.0.
type RPCError struct {
        Code    int    `json:"code"`
        Message string `json:"message"`
}

// RPCResponse — ответ JSON-RPC 2.0.
type RPCResponse struct {
        JSONRPC string          `json:"jsonrpc"`
        ID      json.RawMessage `json:"id"`
        Result  any             `json:"result,omitempty"`
        Error   *RPCError       `json:"error,omitempty"`
}

func rpcFail(id json.RawMessage, code int, msg string) *RPCResponse {
        if len(id) == 0 {
                id = json.RawMessage("null")
        }
        return &RPCResponse{JSONRPC: "2.0", ID: id, Error: &RPCError{Code: code, Message: msg}}
}

// HandleRPC обрабатывает один JSON-RPC запрос. Для уведомлений (без id) возвращает nil.
func (g *Gateway) HandleRPC(ctx context.Context, p domain.Principal, internID string, raw []byte) *RPCResponse {
        var req RPCRequest
        if err := json.Unmarshal(raw, &req); err != nil {
                return rpcFail(nil, rpcParseError, "некорректный JSON")
        }
        if req.JSONRPC != "2.0" || req.Method == "" {
                return rpcFail(req.ID, rpcInvalidRequest, "ожидается JSON-RPC 2.0 с полем method")
        }
        if len(req.ID) == 0 {
                return nil // уведомление (например notifications/initialized) — ответа нет
        }
        ok := func(result any) *RPCResponse { return &RPCResponse{JSONRPC: "2.0", ID: req.ID, Result: result} }

        switch req.Method {
        case "initialize":
                return ok(map[string]any{
                        "protocolVersion": ProtocolVersion,
                        "capabilities":    map[string]any{"tools": map[string]any{"listChanged": false}},
                        "serverInfo":      map[string]any{"name": "avito-start-ai-buddy-mcp", "version": "1.0.0"},
                        "instructions":    "Имитации корпоративных сервисов Авито. Все данные синтетические. Каждый вызов проходит RBAC.",
                })
        case "ping":
                return ok(map[string]any{})
        case "tools/list":
                tools := make([]map[string]any, 0, len(g.reg.order))
                for _, t := range g.reg.Tools() {
                        item := map[string]any{"name": t.Name, "title": t.Title, "description": t.Description, "inputSchema": t.InputSchema}
                        if t.OutputSchema != nil {
                                item["outputSchema"] = t.OutputSchema
                        }
                        tools = append(tools, item)
                }
                return ok(map[string]any{"tools": tools})
        case "tools/call":
                var params struct {
                        Name      string         `json:"name"`
                        Arguments map[string]any `json:"arguments"`
                }
                if err := json.Unmarshal(req.Params, &params); err != nil || params.Name == "" {
                        return rpcFail(req.ID, rpcInvalidParams, "ожидаются params.name и params.arguments")
                }
                call := g.Execute(ctx, p, internID, params.Name, params.Arguments)
                if call.ErrorCode == CodeUnknownTool {
                        return rpcFail(req.ID, rpcInvalidParams, call.Error)
                }
                if !call.OK() {
                        text := fmt.Sprintf("%s: %s", call.ErrorCode, call.Error)
                        return ok(map[string]any{
                                "isError": true,
                                "content": []map[string]any{{"type": "text", "text": text}},
                                "_meta":   map[string]any{"rbac": call.RBAC, "tool_call_id": call.ID},
                        })
                }
                text, _ := json.Marshal(call.Result)
                return ok(map[string]any{
                        "isError":           false,
                        "content":           []map[string]any{{"type": "text", "text": string(text)}},
                        "structuredContent": call.Result,
                        "_meta":             map[string]any{"rbac": call.RBAC, "tool_call_id": call.ID},
                })
        default:
                return rpcFail(req.ID, rpcMethodNotFound, fmt.Sprintf("метод %q не поддерживается", req.Method))
        }
}

// ValidateArgs проверяет аргументы по JSON Schema из спецификации инструмента.
// Поддерживается подмножество, достаточное для контрактов /specs: type, required,
// properties, additionalProperties, enum, pattern, minLength, maxLength, minimum, maximum, items.
func ValidateArgs(schema map[string]any, args map[string]any) []string {
        if args == nil {
                args = map[string]any{}
        }
        var errs []string
        validateValue("", schema, args, &errs)
        sort.Strings(errs)
        return errs
}

var (
        patternMu    sync.Mutex
        patternCache = map[string]*regexp.Regexp{}
)

func compile(pattern string) (*regexp.Regexp, error) {
        patternMu.Lock()
        defer patternMu.Unlock()
        if re, ok := patternCache[pattern]; ok {
                return re, nil
        }
        re, err := regexp.Compile(pattern)
        if err != nil {
                return nil, err
        }
        patternCache[pattern] = re
        return re, nil
}

func toFloat(v any) (float64, bool) {
        switch n := v.(type) {
        case float64:
                return n, true
        case float32:
                return float64(n), true
        case int:
                return float64(n), true
        case int64:
                return float64(n), true
        case json.Number:
                f, err := n.Float64()
                return f, err == nil
        }
        return 0, false
}

func fieldName(path string) string {
        if path == "" {
                return "аргументы"
        }
        return path
}

func join(path, key string) string {
        if path == "" {
                return key
        }
        return path + "." + key
}

func validateValue(path string, schema map[string]any, value any, errs *[]string) {
        if schema == nil {
                return
        }
        typ, _ := schema["type"].(string)
        switch typ {
        case "object":
                obj, ok := value.(map[string]any)
                if !ok {
                        *errs = append(*errs, fmt.Sprintf("%s: ожидается объект", fieldName(path)))
                        return
                }
                props, _ := schema["properties"].(map[string]any)
                if required, ok := schema["required"].([]any); ok {
                        for _, r := range required {
                                name, _ := r.(string)
                                if v, present := obj[name]; !present || v == nil {
                                        *errs = append(*errs, fmt.Sprintf("%s: обязательное поле", join(path, name)))
                                }
                        }
                }
                for key, v := range obj {
                        sub, known := props[key].(map[string]any)
                        if !known {
                                if extra, ok := schema["additionalProperties"].(bool); ok && !extra {
                                        *errs = append(*errs, fmt.Sprintf("%s: неизвестное поле", join(path, key)))
                                }
                                continue
                        }
                        if v == nil {
                                continue
                        }
                        validateValue(join(path, key), sub, v, errs)
                }
        case "string":
                s, ok := value.(string)
                if !ok {
                        *errs = append(*errs, fmt.Sprintf("%s: ожидается строка", fieldName(path)))
                        return
                }
                n := float64(len([]rune(s)))
                if min, ok := toFloat(schema["minLength"]); ok && n < min {
                        *errs = append(*errs, fmt.Sprintf("%s: минимум %d симв.", fieldName(path), int(min)))
                }
                if max, ok := toFloat(schema["maxLength"]); ok && n > max {
                        *errs = append(*errs, fmt.Sprintf("%s: максимум %d симв.", fieldName(path), int(max)))
                }
                if pattern, ok := schema["pattern"].(string); ok {
                        re, err := compile(pattern)
                        if err != nil {
                                *errs = append(*errs, fmt.Sprintf("%s: некорректный pattern в спецификации", fieldName(path)))
                        } else if !re.MatchString(s) {
                                *errs = append(*errs, fmt.Sprintf("%s: значение %q не соответствует формату", fieldName(path), s))
                        }
                }
                checkEnum(path, schema, s, errs)
        case "integer", "number":
                f, ok := toFloat(value)
                if !ok {
                        *errs = append(*errs, fmt.Sprintf("%s: ожидается число", fieldName(path)))
                        return
                }
                if typ == "integer" && f != math.Trunc(f) {
                        *errs = append(*errs, fmt.Sprintf("%s: ожидается целое число", fieldName(path)))
                }
                if min, ok := toFloat(schema["minimum"]); ok && f < min {
                        *errs = append(*errs, fmt.Sprintf("%s: минимум %v", fieldName(path), min))
                }
                if max, ok := toFloat(schema["maximum"]); ok && f > max {
                        *errs = append(*errs, fmt.Sprintf("%s: максимум %v", fieldName(path), max))
                }
        case "boolean":
                if _, ok := value.(bool); !ok {
                        *errs = append(*errs, fmt.Sprintf("%s: ожидается true/false", fieldName(path)))
                }
        case "array":
                var list []any
                switch v := value.(type) {
                case []any:
                        list = v
                case []string:
                        for _, s := range v {
                                list = append(list, s)
                        }
                default:
                        *errs = append(*errs, fmt.Sprintf("%s: ожидается массив", fieldName(path)))
                        return
                }
                if items, ok := schema["items"].(map[string]any); ok {
                        for i, it := range list {
                                validateValue(fmt.Sprintf("%s[%d]", path, i), items, it, errs)
                        }
                }
        }
}

func checkEnum(path string, schema map[string]any, s string, errs *[]string) {
        enum, ok := schema["enum"].([]any)
        if !ok {
                return
        }
        for _, e := range enum {
                if es, _ := e.(string); es == s {
                        return
                }
        }
        *errs = append(*errs, fmt.Sprintf("%s: значение %q не входит в допустимый список", fieldName(path), s))
}

// argString достаёт строковый аргумент.
func argString(args map[string]any, key string) string {
        s, _ := args[key].(string)
        return s
}

// argStrings достаёт массив строк.
func argStrings(args map[string]any, key string) []string {
        switch v := args[key].(type) {
        case []string:
                return append([]string{}, v...)
        case []any:
                out := make([]string, 0, len(v))
                for _, it := range v {
                        if s, ok := it.(string); ok {
                                out = append(out, s)
                        }
                }
                return out
        }
        return nil
}

// argInt достаёт целочисленный аргумент или значение по умолчанию.
func argInt(args map[string]any, key string, def int) int {
        if f, ok := toFloat(args[key]); ok {
                return int(f)
        }
        return def
}

// toMap превращает структуру в map через JSON (результат инструмента).
func toMap(v any) map[string]any {
        raw, err := json.Marshal(v)
        if err != nil {
                return map[string]any{}
        }
        out := map[string]any{}
        if err := json.Unmarshal(raw, &out); err != nil {
                return map[string]any{}
        }
        return out
}

// RequestRepoAccess выдаёт доступ к репозиторию и стенду. Повторный запрос идемпотентен.
func (s *IAM) RequestRepoAccess(ctx context.Context, c Call) (map[string]any, error) {
        internID := argString(c.Args, "intern_id")
        repo := argString(c.Args, "repository")
        if _, err := loadIntern(ctx, s.store, internID); err != nil {
                return nil, err
        }
        grants, err := domain.Load[domain.AccessGrant](ctx, s.store, domain.KindAccessGrant, internID)
        if err != nil {
                return nil, err
        }
        for _, g := range grants {
                if g.Repository == repo {
                        out := toMap(g)
                        out["status"], out["already_granted"] = "granted", true
                        return out, nil
                }
        }
        stand := argString(c.Args, "stand")
        if stand == "" {
                stand = "test"
        }
        level := argString(c.Args, "access_level")
        if level == "" {
                level = "developer"
        }
        grant := domain.AccessGrant{
                ID: s.ids.NewID("grant"), InternID: internID, Repository: repo, AccessLevel: level,
                Stand:       standName(s.repos[repo], stand),
                TokenMasked: domain.MaskToken(syntheticToken()), // сырой токен не сохраняется и не возвращается
                ExpiresAt:   c.Now.AddDate(0, 0, 90), IsSynthetic: true, CreatedAt: c.Now,
        }
        if err := domain.Save(ctx, s.store, domain.KindAccessGrant, grant.ID, internID, c.Now, grant); err != nil {
                return nil, err
        }
        out := toMap(grant)
        out["status"], out["already_granted"] = "granted", false
        return out, nil
}

// Search ищет документы, видимые роли и команде.
// Если заданы темы — кандидатами становятся только документы этих тем (точный контекст интента),
// иначе выполняется поиск по ключевым словам с отсечением слабых совпадений.
func (s *Playbook) Search(role domain.Role, team, query string, topics []string, limit int) []domain.KnowledgeDoc {
        if limit <= 0 {
                limit = 3
        }
        normalized, tokens := domain.Normalize(query), domain.Tokens(query)
        type scored struct {
                doc   domain.KnowledgeDoc
                score int
        }
        var hits []scored
        top := 0
        for _, d := range s.docs {
                if !d.VisibleTo(role, team) || (len(topics) > 0 && !intersects(topics, d.Topics)) {
                        continue
                }
                score := 0
                for _, kw := range d.Keywords {
                        if domain.MatchTerm(kw, normalized, tokens) {
                                score += 2
                        }
                }
                if score > top {
                        top = score
                }
                hits = append(hits, scored{d, score})
        }
        out := make([]domain.KnowledgeDoc, 0, limit)
        sort.SliceStable(hits, func(i, j int) bool { return hits[i].score > hits[j].score })
        for _, h := range hits {
                switch {
                case len(out) == limit:

// CreateDayOff создаёт заявку на отгул, уведомляет наставника и руководителя.
func (s *Absences) CreateDayOff(ctx context.Context, c Call) (map[string]any, error) {
        internID := argString(c.Args, "intern_id")
        intern, err := loadIntern(ctx, s.store, internID)
        if err != nil {
                return nil, err
        }
        req := domain.AbsenceRequest{
                ID: s.ids.NewID("abs"), InternID: internID, InternName: intern.Name,
                Date: argString(c.Args, "date"), TimeFrom: argString(c.Args, "time_from"), TimeTo: argString(c.Args, "time_to"),
                Reason: strings.TrimSpace(argString(c.Args, "reason")), Status: domain.AbsenceStatusPending,
                Notified: []string{}, IsSynthetic: true, CreatedAt: c.Now,
        }
        if err := req.Validate(c.Now); err != nil {
                return nil, err
        }
        all, err := domain.Load[domain.AbsenceRequest](ctx, s.store, domain.KindAbsence, "")
        if err != nil {
                return nil, err
        }
        for _, a := range all {
                if a.InternID == internID && a.Date == req.Date {
                        return nil, Errorf(CodeDuplicate, "заявка на %s уже оформлена: %s", humanDate(req.Date), a.Number)
                }
        }
        year := c.Now.Year()
        if t, err := time.Parse("2006-01-02", req.Date); err == nil {
                year = t.Year()
        }
        req.Number = domain.AbsenceNumber(year, len(all))

        notify := argStrings(c.Args, "notify")
        if len(notify) == 0 {
                notify = []string{"mentor", "team_lead"}
        }
        when := humanDate(req.Date)
        if req.TimeFrom != "" {
                when += fmt.Sprintf(", %s–%s", req.TimeFrom, req.TimeTo)
        }
        recipients := map[string]struct{ id, name, note string }{
                "mentor":    {intern.MentorID, intern.MentorName, "К сведению: учти при планировании встреч и ревью."},
                "team_lead": {intern.TeamLeadID, intern.TeamLeadName, "Заявка ожидает согласования в сервисе «Отсутствия»."},
        }
        for _, key := range notify {
                r, ok := recipients[key]
                if !ok || r.id == "" {
                        continue
                }
                body := fmt.Sprintf("%s оформил(а) отгул на %s. Причина: %s. Заявка %s. %s", intern.Name, when, req.Reason, req.Number, r.note)
                if _, err := s.notifications.Send(ctx, r.id, "Отгул: "+intern.Name, body, "absence", c.Now); err != nil {
                        return nil, err
                }
                req.Notified = append(req.Notified, r.name)
        }
        if err := domain.Save(ctx, s.store, domain.KindAbsence, req.ID, internID, c.Now, req); err != nil {
                return nil, err
        }
        return map[string]any{
                "request_id": req.ID, "request_number": req.Number, "status": req.Status, "date": req.Date,
                "time_from": req.TimeFrom, "time_to": req.TimeTo, "reason": req.Reason,
                "notified": toAny(req.Notified), "is_synthetic": true,
        }, nil
}

// ReviewDayOff согласовывает или отклоняет заявку на отгул руководителем.
func (s *Absences) ReviewDayOff(ctx context.Context, c Call) (map[string]any, error) {
        id := argString(c.Args, "id")
        decision := strings.TrimSpace(argString(c.Args, "decision"))
        comment := strings.TrimSpace(argString(c.Args, "comment"))
        if decision != domain.AbsenceStatusApproved && decision != domain.AbsenceStatusRejected {
                return nil, Errorf(CodeInvalidArgs, "некорректное решение %q: ожидается approved или rejected", decision)
        }
        all, err := domain.Load[domain.AbsenceRequest](ctx, s.store, domain.KindAbsence, "")
        if err != nil {
                return nil, err
        }
        var target *domain.AbsenceRequest
        for i := range all {
                if all[i].ID == id || all[i].Number == id {
                        target = &all[i]
                        break
                }
        }
        if target == nil {
                return nil, Errorf(CodeNotFound, "заявка на отгул %q не найдена", id)
        }
        target.Status = decision
        target.Decision = decision
        target.Comment = comment
        target.ReviewerID = c.Principal.ID
        t := c.Now
        target.ReviewedAt = &t
        if err := domain.Save(ctx, s.store, domain.KindAbsence, target.ID, target.InternID, target.CreatedAt, *target); err != nil {
                return nil, err
        }
        return map[string]any{
                "id":             target.ID,
                "request_number": target.Number,
                "status":         target.Status,
                "decision":       target.Decision,
                "comment":        target.Comment,
                "is_synthetic":   true,
        }, nil
}

// Flags — имитация сервиса feature flags: compile-time значения из сборки,
// runtime-значения переопределяются через кэш (Redis).
type Flags struct {
        cache    domain.Cache
        defaults []domain.FeatureFlag
}

// Ключи флагов AI-Buddy.
const (
        FlagProactiveCron = "ai_buddy.proactive_cron"
        FlagAbsenceCard   = "ai_buddy.absence_action_card"
        FlagLLMAnswers    = "ai_buddy.llm_answers"
        flagCachePrefix   = "ff:"
)

// NewFlags создаёт сервис флагов.
func NewFlags(cache domain.Cache, defaults []domain.FeatureFlag) *Flags {
        return &Flags{cache: cache, defaults: append([]domain.FeatureFlag{}, defaults...)}
}

// List возвращает флаги с учётом runtime-переопределений. Сбой кэша → значения по умолчанию.
func (f *Flags) List(ctx context.Context) []domain.FeatureFlag {
        out := make([]domain.FeatureFlag, 0, len(f.defaults))
        for _, flag := range f.defaults {
                if flag.Kind == domain.FlagRuntime {
                        if v, ok, err := f.cache.Get(ctx, flagCachePrefix+flag.Key); err == nil && ok {
                                flag.Enabled = v == "1"
                        }
                }
                out = append(out, flag)
        }
        return out
}

// Enabled сообщает, включён ли флаг (неизвестный флаг выключен).
func (f *Flags) Enabled(ctx context.Context, key string) bool {
        for _, flag := range f.List(ctx) {
                if flag.Key == key {
                        return flag.Enabled
                }
        }
        return false
}

// Set переключает runtime-флаг.
func (f *Flags) Set(ctx context.Context, key string, enabled bool) (domain.FeatureFlag, error) {
        for _, flag := range f.defaults {
                if flag.Key != key {
                        continue
                }
                if flag.Kind != domain.FlagRuntime {
                        return flag, fmt.Errorf("%w: флаг %s задаётся при сборке (compile-time)", domain.ErrValidation, key)
                }
                value := "0"
                if enabled {
                        value = "1"
                }
                if err := f.cache.Set(ctx, flagCachePrefix+key, value, 0); err != nil {
                        return flag, err
                }
                flag.Enabled = enabled
                return flag, nil
        }
        return domain.FeatureFlag{}, fmt.Errorf("%w: флаг %s", domain.ErrNotFound, key)
}

// Reset сбрасывает runtime-переопределения.
func (f *Flags) Reset(ctx context.Context) error {
        keys := make([]string, 0, len(f.defaults))
        for _, flag := range f.defaults {
                keys = append(keys, flagCachePrefix+flag.Key)
        }
        return f.cache.Del(ctx, keys...)
}

// ListTool — обработчик feature_flags.list.
func (f *Flags) ListTool(ctx context.Context, _ Call) (map[string]any, error) {
        flags := f.List(ctx)
        list := make([]any, 0, len(flags))
        for _, flag := range flags {
                list = append(list, toMap(flag))
        }
        return map[string]any{"flags": list}, nil
}

// Package rbac — чистый движок проверки прав по матрице из specs/03_rbac_matrix.openspec.yaml.
// Модель: default deny, deny-overrides, ABAC-ограничения (self_only, team_scope) поверх ролей.
package rbac

import (
        "fmt"
        "net/http"
        "sort"
        "strings"

        "aibuddy/backend/internal/domain"
        "aibuddy/backend/internal/spec"
)

// Идентификаторы ограничений и служебных политик.
const (
        ConstraintSelfOnly  = "self_only"
        ConstraintTeamScope = "team_scope"
        PolicyDefaultDeny   = "DEFAULT-DENY"
        PolicyUnknownRole   = "UNKNOWN-ROLE"
        effectAllow         = "allow"
        effectDeny          = "deny"
        wildcard            = "*"
)

// Resource — атрибуты ресурса для ABAC-ограничений.
type Resource struct {
        InternID string // владелец данных, над которыми выполняется операция
        Team     string // команда-владелец ресурса
}

// Engine — движок RBAC.
type Engine struct {
        roles      map[string]spec.Role
        roleOrder  []spec.Role
        policies   map[string][]spec.Policy
        operations []string
}

// New строит движок по спецификации.
func New(s spec.RBACSpec) *Engine {
        e := &Engine{roles: map[string]spec.Role{}, policies: map[string][]spec.Policy{}}
        for _, r := range s.Roles {
                e.roles[r.ID] = r
                e.roleOrder = append(e.roleOrder, r)
        }
        for _, p := range s.Policies {
                if _, seen := e.policies[p.Operation]; !seen {
                        e.operations = append(e.operations, p.Operation)
                }
                e.policies[p.Operation] = append(e.policies[p.Operation], p)
        }
        return e
}

func appliesTo(p spec.Policy, role domain.Role) bool {
        for _, r := range p.Roles {
                if r == wildcard || r == string(role) {
                        return true
                }
        }
        return false
}

func forbidden(p domain.Principal, op, policy, reason string) domain.RBACVerdict {
        return domain.RBACVerdict{
                Allowed: false, Code: domain.VerdictForbidden, HTTPStatus: http.StatusForbidden,
                Role: p.Role, Operation: op, PolicyID: policy, Reason: reason,
        }
}

// Check проверяет, может ли субъект выполнить операцию над ресурсом.
func (e *Engine) Check(p domain.Principal, operation string, res Resource) domain.RBACVerdict {
        role, known := e.roles[string(p.Role)]
        if !known {
                return forbidden(p, operation, PolicyUnknownRole, fmt.Sprintf("Роль %q не входит в ролевую модель", p.Role))
        }

        var allows []spec.Policy
        for _, pol := range e.policies[operation] {
                if !appliesTo(pol, p.Role) {
                        continue
                }
                if pol.Effect == effectDeny {
                        reason := pol.Reason
                        if reason == "" {
                                reason = fmt.Sprintf("Операция %s запрещена политикой %s", operation, pol.ID)
                        }
                        v := forbidden(p, operation, pol.ID, reason)
                        v.Incident = pol.Incident
                        v.SafeAlternative = pol.SafeAlternative
                        return v
                }
                allows = append(allows, pol)
        }

        if len(allows) == 0 {
                return forbidden(p, operation, PolicyDefaultDeny,
                        fmt.Sprintf("Для роли «%s» операция %s не разрешена (default deny)", role.Title, operation))
        }

        var failed domain.RBACVerdict
        for _, pol := range allows {
                constraint, reason := firstFailedConstraint(pol.Constraints, p, res)
                if constraint == "" {
                        return domain.RBACVerdict{
                                Allowed: true, Code: domain.VerdictAllowed, HTTPStatus: http.StatusOK,
                                Role: p.Role, Operation: operation, PolicyID: pol.ID,
                                Reason: allowReason(pol, role),
                        }
                }
                failed = forbidden(p, operation, pol.ID, reason)
                failed.Constraint = constraint
        }
        return failed
}

func allowReason(pol spec.Policy, role spec.Role) string {
        if pol.Description != "" {
                return pol.Description
        }
        return fmt.Sprintf("Разрешено политикой %s для роли «%s»", pol.ID, role.Title)
}

func firstFailedConstraint(constraints []string, p domain.Principal, res Resource) (string, string) {
        for _, c := range constraints {
                switch c {
                case ConstraintSelfOnly:
                        if res.InternID == "" || res.InternID != p.ID {
                                return c, "Операцию можно выполнять только над собственными данными (self_only)"
                        }
                case ConstraintTeamScope:
                        if res.Team == "" || !strings.EqualFold(res.Team, p.Team) {
                                return c, fmt.Sprintf("Ресурс не принадлежит команде %s (team_scope)", p.Team)
                        }
                default:
                        return c, fmt.Sprintf("Неизвестное ограничение %q — доступ запрещён", c)
                }
        }
        return "", ""
}

// Can сообщает, разрешена ли операция роли на уровне матрицы (без учёта ABAC-ограничений).
func (e *Engine) Can(role domain.Role, operation string) bool {
        allowed := false
        for _, pol := range e.policies[operation] {
                if !appliesTo(pol, role) {
                        continue
                }
                if pol.Effect == effectDeny {
                        return false
                }
                allowed = true
        }
        return allowed
}

// MatrixRow — строка матрицы «операция × роль» для отображения.
type MatrixRow struct {
        Operation string          `json:"operation"`
        Access    map[string]bool `json:"access"`
}

// Matrix возвращает развёрнутую матрицу прав для ролей-людей и сервисных аккаунтов.
func (e *Engine) Matrix() []MatrixRow {
        ops := append([]string{}, e.operations...)
        sort.Strings(ops)
        rows := make([]MatrixRow, 0, len(ops))
        for _, op := range ops {
                row := MatrixRow{Operation: op, Access: map[string]bool{}}
                for _, r := range e.roleOrder {
                        row.Access[r.ID] = e.Can(domain.Role(r.ID), op)
                }
                rows = append(rows, row)
        }
        return rows
}

// Roles возвращает роли в порядке спецификации.
func (e *Engine) Roles() []spec.Role { return append([]spec.Role{}, e.roleOrder...) }

// RoleTitle возвращает человекочитаемое название роли.
func (e *Engine) RoleTitle(role domain.Role) string {
        if r, ok := e.roles[string(role)]; ok {
                return r.Title
        }
        return string(role)
}

// Package spec загружает и валидирует SDD-спецификации (OpenSpec) из директории /specs.
// Остальные пакеты получают поведение агента, контракты MCP и матрицу RBAC только отсюда.
package spec

import (
        "fmt"
        "io/fs"
        "sort"
        "strings"

        "gopkg.in/yaml.v3"

        "aibuddy/specs"
)

// Metadata — общий заголовок OpenSpec-файла.
type Metadata struct {
        ID          string `yaml:"id" json:"id"`
        Title       string `yaml:"title" json:"title,omitempty"`
        Version     string `yaml:"version" json:"version"`
        IsSynthetic bool   `yaml:"is_synthetic" json:"is_synthetic"`
}

// ContextVariable — контекстная переменная стажёра, доступная агенту.
type ContextVariable struct {
        Name    string `yaml:"name" json:"name"`
        Source  string `yaml:"source" json:"source"`
        Example string `yaml:"example" json:"example"`
}

// Intent — интент Smart Engine с триггерами и планом вызова инструмента.
type Intent struct {
        ID          string     `yaml:"id"`
        Scenario    string     `yaml:"scenario"`
        Priority    int        `yaml:"priority"`
        Description string     `yaml:"description"`
        Tool        string     `yaml:"tool"`
        ActionCard  string     `yaml:"action_card"`
        ConfirmTool string     `yaml:"confirm_tool"`
        Topics      []string   `yaml:"topics"`
        Triggers    [][]string `yaml:"triggers"`
        Suggestions []string   `yaml:"suggestions"`
}

// ChecklistEffect — правило автоматического обновления чек-листа.
type ChecklistEffect struct {
        Tool     string   `yaml:"tool"`
        DocsRead []string `yaml:"docs_read"`
        Item     string   `yaml:"item"`
        Status   string   `yaml:"status"`
}

// MetricsSpec — параметры оценки экономии времени наставника.
type MetricsSpec struct {
        MentorHoursBaselineMin int            `yaml:"mentor_hours_baseline_min"`
        MentorHoursBaselineMax int            `yaml:"mentor_hours_baseline_max"`
        MinutesSaved           map[string]int `yaml:"minutes_saved"`
}

// Scenario — приёмочный сценарий Given/When/Then.
type Scenario struct {
        ID    string   `yaml:"id"`
        Title string   `yaml:"title"`
        Given string   `yaml:"given"`
        When  string   `yaml:"when"`
        Then  []string `yaml:"then"`
}

// AgentSpec — спецификация поведения агента (01_agent_behavior).
type AgentSpec struct {
        OpenSpec string   `yaml:"openspec"`
        Kind     string   `yaml:"kind"`
        Metadata Metadata `yaml:"metadata"`
        Agent    struct {
                Name             string            `yaml:"name"`
                Language         string            `yaml:"language"`
                Tone             string            `yaml:"tone"`
                SystemPrompt     string            `yaml:"system_prompt"`
                ContextVariables []ContextVariable `yaml:"context_variables"`
                SyntheticData    struct {
                        Disclaimer string   `yaml:"disclaimer"`
                        Rules      []string `yaml:"rules"`
                } `yaml:"synthetic_data"`
                Engine struct {
                        SmartDelayMS int `yaml:"smart_delay_ms"`
                        LLM          struct {
                                Protocol  string   `yaml:"protocol"`
                                Env       []string `yaml:"env"`
                                TimeoutMS int      `yaml:"timeout_ms"`
                                Fallback  string   `yaml:"fallback"`
                        } `yaml:"llm"`
                } `yaml:"engine"`
        } `yaml:"agent"`
        Intents          []Intent          `yaml:"intents"`
        FallbackIntent   Intent            `yaml:"fallback_intent"`
        ChecklistEffects []ChecklistEffect `yaml:"checklist_effects"`
        Responses        map[string]string `yaml:"responses"`
        Metrics          MetricsSpec       `yaml:"metrics"`
        Scenarios        []Scenario        `yaml:"scenarios"`
}

// Response возвращает шаблон ответа с подстановкой {ключ} → значение.

        text := strings.TrimSpace(a.Responses[key])

        }

}

// LoadFS читает и валидирует спецификации из произвольной файловой системы.
func LoadFS(fsys fs.FS) (*Bundle, error) {
        var b Bundle
        files := []struct {
                name string
                kind string
                dst  any
        }{
                {specs.AgentBehaviorFile, "AgentBehavior", &b.Agent},
                {specs.MCPToolsFile, "McpTools", &b.Tools},
                {specs.RBACMatrixFile, "RbacMatrix", &b.RBAC},
        }
        for _, f := range files {
                raw, err := fs.ReadFile(fsys, f.name)
                if err != nil {
                        return nil, fmt.Errorf("spec: чтение %s: %w", f.name, err)
                }
                if err := yaml.Unmarshal(raw, f.dst); err != nil {
                        return nil, fmt.Errorf("spec: разбор %s: %w", f.name, err)
                }
        }
        if b.Agent.Kind != "AgentBehavior" || b.Tools.Kind != "McpTools" || b.RBAC.Kind != "RbacMatrix" {
                return nil, fmt.Errorf("spec: неверный kind (%q, %q, %q)", b.Agent.Kind, b.Tools.Kind, b.RBAC.Kind)
        }
        if err := b.Validate(); err != nil {
                return nil, err
        }
        sort.SliceStable(b.Agent.Intents, func(i, j int) bool {
                return b.Agent.Intents[i].Priority > b.Agent.Intents[j].Priority
        })
        return &b, nil
}

// Validate проверяет целостность и перекрёстные ссылки между спецификациями.
func (b *Bundle) Validate() error {
        var problems []string
        add := func(format string, args ...any) { problems = append(problems, fmt.Sprintf(format, args...)) }

        if strings.TrimSpace(b.Agent.Agent.SystemPrompt) == "" {
                add("agent.system_prompt пуст")
        }
        if strings.TrimSpace(b.Agent.Agent.SyntheticData.Disclaimer) == "" {
                add("agent.synthetic_data.disclaimer пуст")
        }

        servers := map[string]bool{}
        for _, s := range b.Tools.Servers {
                servers[s.ID] = true
        }
        tools := map[string]bool{}
        for _, t := range b.Tools.Tools {
                if tools[t.Name] {
                        add("инструмент %s объявлен дважды", t.Name)
                }
                tools[t.Name] = true
                if !servers[t.Server] {
                        add("инструмент %s ссылается на неизвестный сервер %q", t.Name, t.Server)
                }
                if !strings.HasPrefix(t.Name, t.Server+".") {
                        add("имя инструмента %s должно начинаться с %s.", t.Name, t.Server)
                }
                if t.InputSchema == nil || t.InputSchema["type"] != "object" {
                        add("инструмент %s: input_schema.type должен быть object", t.Name)
                }
        }
        if len(tools) == 0 {
                add("не объявлено ни одного MCP-инструмента")
        }

        intents := map[string]bool{}
        for _, in := range append(append([]Intent{}, b.Agent.Intents...), b.Agent.FallbackIntent) {
                if in.ID == "" {
                        add("интент без id")
                        continue
                }
                if intents[in.ID] {
                        add("интент %s объявлен дважды", in.ID)
                }
                intents[in.ID] = true
                for _, ref := range []string{in.Tool, in.ConfirmTool} {
                        if ref != "" && !tools[ref] {
                                add("интент %s ссылается на неизвестный инструмент %q", in.ID, ref)
                        }
                }
        }
        for _, e := range b.Agent.ChecklistEffects {
                if e.Tool != "" && !tools[e.Tool] {
                        add("checklist_effects: неизвестный инструмент %q", e.Tool)
                }
                if e.Item == "" || e.Status == "" {
                        add("checklist_effects: пустые item/status")
                }
        }

        roles := map[string]bool{}
        for _, r := range b.RBAC.Roles {
                roles[r.ID] = true
        }
        constraints := map[string]bool{}
        for _, c := range b.RBAC.Constraints {
                constraints[c.ID] = true
        }
        if b.RBAC.DefaultEffect != "deny" {
                add("rbac.default_effect должен быть deny")
        }
        covered := map[string]bool{}
        ids := map[string]bool{}
        for _, p := range b.RBAC.Policies {
                if p.ID == "" || ids[p.ID] {
                        add("rbac: политика с пустым или повторным id %q", p.ID)
                }
                ids[p.ID] = true
                if p.Effect != "allow" && p.Effect != "deny" {
                        add("rbac: политика %s: effect %q", p.ID, p.Effect)
                }
                for _, r := range p.Roles {
                        if r != "*" && !roles[r] {
                                add("rbac: политика %s: неизвестная роль %q", p.ID, r)
                        }
                }
                for _, c := range p.Constraints {
                        if !constraints[c] {
                                add("rbac: политика %s: неизвестное ограничение %q", p.ID, c)
                        }
                }
                if !strings.HasPrefix(p.Operation, "api.") && !tools[p.Operation] {
                        add("rbac: политика %s: неизвестная операция %q", p.ID, p.Operation)
                }
                covered[p.Operation] = true
        }
        for name := range tools {
                if !covered[name] {
                        add("rbac: для инструмента %s нет ни одной политики", name)
                }
        }

        if len(problems) > 0 {
                sort.Strings(problems)
                return fmt.Errorf("spec: спецификации невалидны:\n- %s", strings.Join(problems, "\n- "))
        }
        return nil
}

// New создаёт пустое хранилище.

        return &Store{data: map[domain.Kind][]domain.Record{}}
}

// Put создаёт или обновляет запись, сохраняя позицию при обновлении.
func (s *Store) Put(_ context.Context, rec domain.Record) error {
        s.mu.Lock()
        defer s.mu.Unlock()
        rec.Payload = append([]byte(nil), rec.Payload...)
        list := s.data[rec.Kind]
        for i := range list {
                if list[i].ID == rec.ID {
                        rec.CreatedAt = list[i].CreatedAt
                        list[i] = rec
                        return nil
                }
        }
        s.data[rec.Kind] = append(list, rec)
        return nil
}

// Get возвращает запись или domain.ErrNotFound.
func (s *Store) Get(_ context.Context, kind domain.Kind, id string) (domain.Record, error) {
        s.mu.RLock()
        defer s.mu.RUnlock()
        for _, r := range s.data[kind] {
                if r.ID == id {
                        return r, nil
                }
        }
        return domain.Record{}, domain.ErrNotFound
}

// Open подключается к PostgreSQL, проверяет соединение и применяет миграции.
func Open(ctx context.Context, dsn string) (*Store, error) {
        db, err := sql.Open("pgx", dsn)
        if err != nil {
                return nil, fmt.Errorf("postgres: открытие: %w", err)
        }
        db.SetMaxOpenConns(10)
        db.SetConnMaxIdleTime(5 * time.Minute)
        s := New(db)
        if err := s.Ping(ctx); err != nil {
                _ = db.Close()
                return nil, fmt.Errorf("postgres: недоступен: %w", err)
        }
        if err := s.Migrate(ctx); err != nil {
                _ = db.Close()
                return nil, err
        }
        return s, nil
}

// New оборачивает готовое соединение (используется в тестах с sqlmock).
func New(db *sql.DB) *Store { return &Store{db: db} }

// tables — допустимые таблицы (защита от подстановки имени таблицы).
var tables = func() map[domain.Kind]string {
        m := map[domain.Kind]string{}
        for _, k := range domain.AllKinds {
                m[k] = string(k)
        }
        return m
}()

func table(kind domain.Kind) (string, error) {
        t, ok := tables[kind]
        if !ok {
                return "", fmt.Errorf("postgres: неизвестный вид записи %q", kind)
        }
        return t, nil
}

// MigrationStatements возвращает DDL автоматической миграции (идемпотентный).
func MigrationStatements() []string {
        stmts := make([]string, 0, len(domain.AllKinds)*2+2)
        for _, k := range domain.AllKinds {
                t := string(k)
                stmts = append(stmts,
                        fmt.Sprintf(`CREATE TABLE IF NOT EXISTS %s (
        seq        BIGSERIAL,
        id         TEXT PRIMARY KEY,
        owner_id   TEXT NOT NULL,
        created_at TIMESTAMPTZ NOT NULL,
        payload    JSONB NOT NULL
)`, t),
                        fmt.Sprintf(`CREATE INDEX IF NOT EXISTS %s_owner_seq_idx ON %s (owner_id, seq)`, t, t),
                )
        }
        // Аналитические индексы по статусам: чек-лист и заявки на отгул.
        stmts = append(stmts,
                `CREATE INDEX IF NOT EXISTS checklist_items_status_idx ON checklist_items ((payload->>'status'))`,
                `CREATE INDEX IF NOT EXISTS absence_requests_status_idx ON absence_requests ((payload->>'status'))`,
        )
        return stmts
}

// Migrate применяет миграции схемы.
func (s *Store) Migrate(ctx context.Context) error {
        for _, stmt := range MigrationStatements() {
                if _, err := s.db.ExecContext(ctx, stmt); err != nil {
                        return fmt.Errorf("postgres: миграция: %w", err)
                }
        }
        return nil
}

// Put создаёт или обновляет запись.
func (s *Store) Put(ctx context.Context, rec domain.Record) error {
        t, err := table(rec.Kind)
        if err != nil {
                return err
        }
        _, err = s.db.ExecContext(ctx,
                `INSERT INTO `+t+` (id, owner_id, created_at, payload) VALUES ($1, $2, $3, $4::jsonb)
ON CONFLICT (id) DO UPDATE SET owner_id = EXCLUDED.owner_id, payload = EXCLUDED.payload`,
                rec.ID, rec.OwnerID, rec.CreatedAt, string(rec.Payload))
        if err != nil {
                return fmt.Errorf("postgres: запись %s/%s: %w", t, rec.ID, err)
        }
        return nil
}

// Get возвращает запись или domain.ErrNotFound.
func (s *Store) Get(ctx context.Context, kind domain.Kind, id string) (domain.Record, error) {
        t, err := table(kind)
        if err != nil {
                return domain.Record{}, err
        }
        rec := domain.Record{Kind: kind}
        var payload string
        err = s.db.QueryRowContext(ctx,
                `SELECT id, owner_id, created_at, payload::text FROM `+t+` WHERE id = $1`, id).
                Scan(&rec.ID, &rec.OwnerID, &rec.CreatedAt, &payload)
        if errors.Is(err, sql.ErrNoRows) {
                return domain.Record{}, domain.ErrNotFound
        }
        if err != nil {
                return domain.Record{}, fmt.Errorf("postgres: чтение %s/%s: %w", t, id, err)
        }
        rec.Payload = []byte(payload)
        return rec, nil
}

// List возвращает записи владельца в порядке вставки.
func (s *Store) List(ctx context.Context, kind domain.Kind, ownerID string) ([]domain.Record, error) {
        t, err := table(kind)
        if err != nil {
                return nil, err
        }
        query := `SELECT id, owner_id, created_at, payload::text FROM ` + t
        var args []any
        if ownerID != "" {
                query += ` WHERE owner_id = $1`
                args = append(args, ownerID)
        }
        query += ` ORDER BY seq`
        rows, err := s.db.QueryContext(ctx, query, args...)
        if err != nil {
                return nil, fmt.Errorf("postgres: список %s: %w", t, err)
        }
        defer rows.Close()
        out := []domain.Record{}
        for rows.Next() {
                rec := domain.Record{Kind: kind}
                var payload string
                if err := rows.Scan(&rec.ID, &rec.OwnerID, &rec.CreatedAt, &payload); err != nil {
                        return nil, fmt.Errorf("postgres: чтение строки %s: %w", t, err)
                }
                rec.Payload = []byte(payload)
                out = append(out, rec)
        }
        if err := rows.Err(); err != nil {
                return nil, fmt.Errorf("postgres: обход %s: %w", t, err)
        }
        return out, nil
}

// Reset атомарно очищает все таблицы и загружает seed.
func (s *Store) Reset(ctx context.Context, seed []domain.Record) error {
        tx, err := s.db.BeginTx(ctx, nil)
        if err != nil {
                return fmt.Errorf("postgres: начало транзакции: %w", err)
        }
        defer func() { _ = tx.Rollback() }()

        names := make([]string, 0, len(domain.AllKinds))
        for _, k := range domain.AllKinds {
                names = append(names, string(k))
        }
        if _, err := tx.ExecContext(ctx, `TRUNCATE TABLE `+strings.Join(names, ", ")+` RESTART IDENTITY`); err != nil {
                return fmt.Errorf("postgres: очистка: %w", err)
        }
        for _, rec := range seed {
                t, err := table(rec.Kind)
                if err != nil {
                        return err
                }
                if _, err := tx.ExecContext(ctx,
                        `INSERT INTO `+t+` (id, owner_id, created_at, payload) VALUES ($1, $2, $3, $4::jsonb)`,
                        rec.ID, rec.OwnerID, rec.CreatedAt, string(rec.Payload)); err != nil {
                        return fmt.Errorf("postgres: seed %s/%s: %w", t, rec.ID, err)
                }
        }
        if err := tx.Commit(); err != nil {
                return fmt.Errorf("postgres: фиксация seed: %w", err)
        }
        return nil
}

// Execute выполняет MCP-инструмент по прямому подтверждению из интерфейса
// (например, отправка формы отгула). Права проверяет RBAC-шлюз по роли вызывающего.
func (s *Service) Execute(ctx context.Context, p domain.Principal, tool string, args map[string]any, messageID string, targetInternID ...string) (ChatResult, error) {
        started := time.Now()
        var res ChatResult
        if strings.TrimSpace(tool) == "" {
                return res, fmt.Errorf("%w: не указан инструмент", domain.ErrValidation)
        }
        internID := s.resolveInternID(p, targetInternID...)
        s.Events.Track(ctx, internID, p, domain.EventKindClickstream, "ui.action_confirmed", domain.SeverityInfo,
                map[string]any{"tool": tool, "message_id": messageID})

        call := s.Gateway.Execute(ctx, p, internID, tool, args)
        verdict := call.RBAC
        res = ChatResult{ToolCalls: []domain.MCPToolCall{call}, RBAC: &verdict, ChecklistUpdates: []domain.ChecklistItem{}}

        var updates []domain.ChecklistItem
        if call.OK() {
                var err error
                if updates, err = s.applyEffects(ctx, p, res.ToolCalls, nil, internID); err != nil {
                        return res, err
                }
                res.ChecklistUpdates = nonNil(updates)
        }
        checklist, err := domain.LoadChecklist(ctx, s.Store, internID)
        if err != nil {
                return res, err
        }
        res.Checklist, res.Progress, res.NextStep = checklist, domain.ComputeProgress(checklist), domain.NextStep(checklist)

        if call.OK() && p.ID == internID {
                if messageID != "" {
                        s.markCardSubmitted(ctx, messageID, internID)
                }
                text, card, saving := s.describeAction(call)
                msg := domain.ChatMessage{
                        ID: s.IDs.NewID("msg"), InternID: internID, Role: domain.MessageAssistant,
                        Text: text + checklistNote(updates, checklist), Intent: "action_confirmed", Engine: domain.EngineSmart,
                        Sources: []domain.DocRef{}, ResultCard: card, Suggestions: []string{"Что дальше по чек-листу?"},
                        ToolCallIDs: []string{call.ID}, CreatedAt: s.Clock.Now(),
                }
                if err := s.saveMessage(ctx, msg); err != nil {
                        return res, err
                }
                res.Message = &msg
                payload := map[string]any{"tool": tool, "engine": domain.EngineSmart, "answered": true}
                if saving != "" {
                        payload["saving_key"] = saving
                }
                s.Events.Track(ctx, internID, p, domain.EventKindAgent, "agent.action", domain.SeverityInfo, payload)
        }
        res.LatencyMS = float64(time.Since(started).Microseconds()) / 1000
        return res, nil
}

// markCardSubmitted помечает карточку исходного сообщения как отправленную (best-effort).

        msg, err := domain.LoadOne[domain.ChatMessage](ctx, s.Store, domain.KindMessage, messageID)
        if err != nil || msg.InternID != internID || msg.ActionCard == nil {
                return
        }
        msg.ActionCard.Status = domain.ActionStatusDone
        _ = s.saveMessage(ctx, msg)
}

// describeAction формирует подтверждение и карточку результата по итогам инструмента.

        r := call.Result
        switch call.Tool {
        case mcp.ToolCreateDayOff:
                when := agent.HumanDay(str(r, "date"))
                if from := str(r, "time_from"); from != "" {
                        when += fmt.Sprintf(", %s–%s", from, str(r, "time_to"))

func sameStrings(a, b []string) bool {
        if len(a) != len(b) {
                return false
        }
        for i := range a {
                if a[i] != b[i] {
                        return false
                }
        }
        return true
}

// CronTrigger имитирует запуск фонового ИИ-крона Авито: если первая задача стажёра висит
// в работе без коммитов, отправляет стажёру проактивное сообщение и готовит наставнику
// Structured 1:1 Brief. Джоб исполняется от сервисного аккаунта system_cron.
func (s *Service) CronTrigger(ctx context.Context, p domain.Principal) (CronResult, error) {
        res := CronResult{ToolCalls: []domain.MCPToolCall{}}
        if err := s.authorize(ctx, p, OpCronTrigger); err != nil {
                return res, err
        }
        cronEmployee, ok := s.Directory.ByID(s.CronID)
        if !ok {
                return res, fmt.Errorf("сервисный аккаунт крона %q не найден", s.CronID)
        }
        cron := cronEmployee.Principal()

        finish := func(reason string) (CronResult, error) {
                res.Reason = reason
                checklist, err := domain.LoadChecklist(ctx, s.Store, s.internID)
                if err != nil {
                        return res, err
                }
                res.Checklist, res.Progress = checklist, domain.ComputeProgress(checklist)
                payload := map[string]any{
                        "triggered": res.Triggered, "deduplicated": res.Deduplicated, "reason": reason, "requested_by": p.ID,
                }
                if res.Brief != nil && !res.Deduplicated {
                        payload["brief_id"], payload["saving_key"] = res.Brief.ID, domain.SavingBrief
                }
                s.Events.Track(ctx, s.internID, cron, domain.EventKindCron, "cron.run", domain.SeverityInfo, payload)
                return res, nil
        }

        if !s.Services.Flags.Enabled(ctx, mcp.FlagProactiveCron) {
                return finish("Feature flag " + mcp.FlagProactiveCron + " выключен — крон пропускает запуск")
        }

        activity := s.Gateway.Execute(ctx, cron, s.internID, mcp.ToolTaskActivity, map[string]any{"intern_id": s.internID})
        res.ToolCalls = append(res.ToolCalls, activity)
        if !activity.OK() {
                return finish("Трекер задач недоступен: " + activity.Error)
        }
        task, err := domain.LoadOne[domain.SprintTask](ctx, s.Store, domain.KindSprintTask, s.internID)
        if err != nil {
                return res, err
        }
        res.Task = &task
        if !task.Stalled() {
                return finish(fmt.Sprintf("Задача %s не требует внимания: статус %s, коммитов %d", task.Key, task.Status, task.Commits))
        }

        intern, err := domain.LoadOne[domain.InternProfile](ctx, s.Store, domain.KindIntern, s.internID)
        if err != nil {
                return res, err
        }
        learned, err := s.learnedTopics(ctx)
        if err != nil {
                return res, err
        }
        open, err := s.openQuestions(ctx)
        if err != nil {
                return res, err
        }
        briefs, err := domain.Load[domain.MentorBrief](ctx, s.Store, domain.KindBrief, s.internID)
        if err != nil {
                return res, err
        }
        res.Triggered = true
        if n := len(briefs); n > 0 {
                last := briefs[n-1]
                if last.TaskKey == task.Key && sameStrings(last.LearnedTopics, learned) && sameStrings(last.OpenQuestions, open) {
                        res.Deduplicated, res.Brief = true, &last
                        return finish("Бриф по задаче уже отправлен наставнику, новых данных нет — повторно не беспокоим")
                }
        }

        checklist, err := domain.LoadChecklist(ctx, s.Store, s.internID)
        if err != nil {
                return res, err
        }
        progress := domain.ComputeProgress(checklist)
        signals := []string{
                fmt.Sprintf("Задача %s «%s» в статусе in_progress %d дн., коммитов: %d", task.Key, task.Title, task.DaysInProgress, task.Commits),
                fmt.Sprintf("Прогресс чек-листа адаптации: %d из %d шагов", progress.Completed, progress.Total),
        }
        briefCall := s.Gateway.Execute(ctx, cron, s.internID, mcp.ToolSendBrief, map[string]any{
                "intern_id": s.internID, "mentor_id": intern.MentorID, "task_key": task.Key,
                "learned_topics": toAny(learned), "architecture_question": task.ArchitectureQuestion,
                "signals": toAny(signals), "open_questions": toAny(open),
        })
        res.ToolCalls = append(res.ToolCalls, briefCall)
        if !briefCall.OK() {
                res.Triggered = false
                return finish("Не удалось отправить бриф наставнику: " + briefCall.Error)
        }
        brief, err := domain.LoadOne[domain.MentorBrief](ctx, s.Store, domain.KindBrief, str(briefCall.Result, "brief_id"))
        if err != nil {
                return res, err
        }
        res.Brief = &brief

        msg := domain.ChatMessage{
                ID: s.IDs.NewID("msg"), InternID: s.internID, Role: domain.MessageAssistant, Proactive: true,
                Intent: "proactive_checkin", Engine: domain.EngineCron,
                Text: fmt.Sprintf("%s, привет! Вижу, что задача **%s** («%s») в работе уже %d-й день, а коммитов пока нет. "+
                        "На первой неделе это нормально — давай помогу сдвинуться:\n"+
                        "- подскажу, с чего начать и как разбить задачу на шаги;\n"+
                        "- напомню, как катить изменения под фича-флагом и проходить ревью.\n\n"+
                        "Архитектурный вопрос по задаче я уже передал наставнику (%s) в брифе к 1:1 — он придёт на встречу подготовленным.",
                        intern.FirstName(), task.Key, task.Summary, task.DaysInProgress, intern.MentorName),
                Sources: []domain.DocRef{},
                Suggestions: []string{
                        agent.ScenarioPrompts["first_task"],
                        agent.ScenarioPrompts["team_processes"],
                },
                ToolCallIDs: callIDs(res.ToolCalls), CreatedAt: s.Clock.Now(),
        }
        if err := s.saveMessage(ctx, msg); err != nil {
                return res, err
        }
        res.ProactiveMessage = &msg
        if _, err := s.applyEffects(ctx, cron, res.ToolCalls, nil); err != nil {
                return res, err
        }
        return finish(fmt.Sprintf("Задача %s без коммитов %d дн.: стажёру отправлено проактивное сообщение, наставнику — бриф к 1:1",
                task.Key, task.DaysInProgress))
}

// TelemetryView — данные инспектора телеметрии (MCP + RBAC + Clickstream).
type TelemetryView struct {
        Events    []domain.TelemetryEvent `json:"events"`
        ToolCalls []domain.MCPToolCall    `json:"tool_calls"`
        Tools     []ToolInfo              `json:"tools"`
        Servers   []spec.Server           `json:"servers"`
        Roles     []spec.Role             `json:"roles"`
        Matrix    []rbac.MatrixRow        `json:"matrix"`
}

const telemetryLimit = 200

func latest[T any](in []T, limit int) []T {
        out := reverse(in)
        if len(out) > limit {
                out = out[:limit]
        }
        return out
}

// Telemetry возвращает журнал событий, вызовов MCP и матрицу RBAC для инспектора.
func (s *Service) Telemetry(ctx context.Context, p domain.Principal) (TelemetryView, error) {
        var v TelemetryView
        if err := s.authorize(ctx, p, OpTelemetryRead); err != nil {
                return v, err
        }
        events, err := domain.Load[domain.TelemetryEvent](ctx, s.Store, domain.KindEvent, s.internID)
        if err != nil {
                return v, err
        }
        calls, err := domain.Load[domain.MCPToolCall](ctx, s.Store, domain.KindToolCall, s.internID)
        if err != nil {
                return v, err
        }
        reg := s.Gateway.Registry()
        tools := make([]ToolInfo, 0, len(reg.Tools()))
        for _, t := range reg.Tools() {
                tools = append(tools, ToolInfo{Tool: t, AllowedForViewer: s.RBAC.Can(p.Role, t.Name)})
        }
        return TelemetryView{
                Events: latest(events, telemetryLimit), ToolCalls: latest(calls, telemetryLimit),
                Tools: tools, Servers: reg.Servers(), Roles: s.RBAC.Roles(), Matrix: s.RBAC.Matrix(),
        }, nil
}

var uiEventRe = regexp.MustCompile(`^ui\.[a-z0-9_.]{1,48}$`)

// TrackUI записывает clickstream-событие интерфейса (имя вида ui.*; payload — плоский).
func (s *Service) TrackUI(ctx context.Context, p domain.Principal, name string, payload map[string]any) (domain.TelemetryEvent, error) {
        if !uiEventRe.MatchString(name) {
                return domain.TelemetryEvent{}, fmt.Errorf("%w: имя события должно иметь вид ui.<название>", domain.ErrValidation)
        }
        clean := map[string]any{}
        for k, v := range payload {
                if len(clean) >= 10 {
                        break
                }
                switch t := v.(type) {
                case string:
                        if len([]rune(t)) > 200 {
                                t = string([]rune(t)[:200])
                        }
                        clean[k] = t
                case float64, bool, int:
                        clean[k] = t
                }
        }
        return s.Events.Track(ctx, s.internID, p, domain.EventKindClickstream, name, domain.SeverityInfo, clean), nil
}

        return s.internID

        if raw, ok, err := s.Cache.Get(ctx, s.sessionKey(internID)); err == nil && ok {

        }
        return sess
}

        if raw, err := json.Marshal(sess); err == nil {
                _ = s.Cache.Set(ctx, s.sessionKey(internID), string(raw), sessionTTL)
        }
}

// Authenticate находит сотрудника по демо-токену (имитация корпоративной авторизации).

        e, ok := s.Directory.ByToken(strings.TrimSpace(token))
        if !ok || token == "" {
                return domain.Principal{}, ErrUnauthorized
        }
        return e.Principal(), nil
}

// authorize проверяет операцию API по матрице RBAC и журналирует отказ.
func (s *Service) authorize(ctx context.Context, p domain.Principal, op string) error {
        v := s.RBAC.Check(p, op, rbac.Resource{})
        if v.Allowed {
                return nil
        }
        s.Events.Track(ctx, s.internID, p, domain.EventKindRBAC, "rbac.decision", domain.SeverityWarn, map[string]any{
                "operation": v.Operation, "role": string(v.Role), "allowed": false, "code": v.Code,
                "policy_id": v.PolicyID, "reason": v.Reason,
        })
        return &ForbiddenError{Verdict: v}
}

func (s *Service) can(p domain.Principal, op string) bool {
        return s.RBAC.Check(p, op, rbac.Resource{}).Allowed
}

// seedRecords — исходные записи: seed + приветственное сообщение агента.
func (s *Service) seedRecords() ([]domain.Record, error) {
        now := s.Clock.Now()
        recs, err := s.Seed.Records(now)
        if err != nil {
                return nil, err
        }
        interns := s.Seed.Interns
        if len(interns) == 0 {
                interns = []domain.InternProfile{s.Seed.Intern}
        }
        for _, in := range interns {
                greeting, _ := s.Spec.Agent.IntentByID(agent.IntentGreeting)
                msgID := welcomeMessage
                if in.ID != s.internID {
                        msgID = welcomeMessage + "-" + in.ID
                }
                suggestions := append([]string{}, greeting.Suggestions...)
                if in.ID == seed.Intern2ID {
                        suggestions = []string{
                                "Как устроены ML-пайплайны в команде: DVC, Airflow и версионирование данных?",
                                "Как в нашей команде катить ML-модель под фича-флагом и запустить A/B-тест?",
                                "Что дальше по чек-листу?",
                        }
                }
                welcome := domain.ChatMessage{
                        ID: msgID, InternID: in.ID, Role: domain.MessageAssistant,
                        Text:   s.Spec.Agent.Response("greeting", map[string]string{"name": in.FirstName(), "team": in.Team}),
                        Intent: agent.IntentGreeting, Engine: domain.EngineSmart,
                        Sources: []domain.DocRef{}, Suggestions: suggestions,
                        ToolCallIDs: []string{}, CreatedAt: now,
                }
                rec, err := domain.NewRecord(domain.KindMessage, welcome.ID, in.ID, now, welcome)
                if err != nil {
                        return nil, err
                }
                recs = append(recs, rec)
        }
        return recs, nil
}

// EnsureSeed загружает синтетические данные при первом старте (пустое хранилище).
func (s *Service) EnsureSeed(ctx context.Context) error {
        _, err := s.Store.Get(ctx, domain.KindIntern, s.internID)
        if err == nil {
                return nil
        }
        if !errors.Is(err, domain.ErrNotFound) {
                return err
        }
        return s.resetData(ctx)
}

func (s *Service) resetData(ctx context.Context) error {
        recs, err := s.seedRecords()
        if err != nil {
                return err
        }
        if err := s.Store.Reset(ctx, recs); err != nil {
                return err
        }
        _ = s.Services.Flags.Reset(ctx)
        _ = s.Cache.Del(ctx, s.sessionKey(s.internID))
        for _, in := range s.Seed.Interns {
                _ = s.Cache.Del(ctx, s.sessionKey(in.ID))
        }
        return nil
}

        Read  []string `json:"read"`
        Total int      `json:"total"`
}

        Mode  string `json:"mode"`
        Model string `json:"model"`
}

// StorageInfo — активные реализации хранилища и кэша.

// WorkspaceView — состояние рабочего пространства для роли смотрящего.
type WorkspaceView struct {
        Viewer        ViewerInfo              `json:"viewer"`
        Intern        domain.InternProfile    `json:"intern"`
        Checklist     []domain.ChecklistItem  `json:"checklist"`
        Progress      domain.Progress         `json:"progress"`
        NextStep      *domain.ChecklistItem   `json:"next_step"`
        SprintTask    domain.SprintTask       `json:"sprint_task"`
        FeatureFlags  []domain.FeatureFlag    `json:"feature_flags"`
        Metrics       domain.MetricsSummary   `json:"metrics"`
        Messages      []domain.ChatMessage    `json:"messages"`
        Notifications []domain.Notification   `json:"notifications"`
        Briefs        []domain.MentorBrief    `json:"briefs"`
        Absences      []domain.AbsenceRequest `json:"absences"`
        AccessGrants  []domain.AccessGrant    `json:"access_grants"`
        Incidents     []domain.TelemetryEvent `json:"incidents"`
        DocProgress   DocProgress             `json:"doc_progress"`
        Engine        EngineInfo              `json:"engine"`
        Storage       StorageInfo             `json:"storage"`
        Permissions   map[string]bool         `json:"permissions"`
        Disclaimer    string                  `json:"disclaimer"`
}

type docRead struct {
        DocID  string    `json:"doc_id"`
        ReadAt time.Time `json:"read_at"`
}

type openQuestion struct {
        Text    string    `json:"text"`
        AskedAt time.Time `json:"asked_at"`
}

func (s *Service) docsRead(ctx context.Context) ([]string, error) {
        return s.docsReadFor(ctx, s.internID)
}

func (s *Service) docsReadFor(ctx context.Context, internID string) ([]string, error) {
        if internID == "" {
                internID = s.internID
        }
        reads, err := domain.Load[docRead](ctx, s.Store, domain.KindDocRead, internID)
        if err != nil {
                return nil, err
        }
        ids := make([]string, 0, len(reads))
        for _, r := range reads {
                ids = append(ids, r.DocID)
        }
        return ids, nil
}

func (s *Service) openQuestions(ctx context.Context) ([]string, error) {
        qs, err := domain.Load[openQuestion](ctx, s.Store, domain.KindOpenQuestion, s.internID)
        if err != nil {
                return nil, err
        }
        out := make([]string, 0, len(qs))
        for _, q := range qs {
                out = append(out, q.Text)
        }
        return out, nil
}

func (s *Service) metrics(ctx context.Context, progress domain.Progress) (domain.MetricsSummary, error) {
        events, err := domain.Load[domain.TelemetryEvent](ctx, s.Store, domain.KindEvent, s.internID)
        if err != nil {
                return domain.MetricsSummary{}, err
        }
        read, err := s.docsRead(ctx)
        if err != nil {
                return domain.MetricsSummary{}, err
        }
        open, err := s.openQuestions(ctx)
        if err != nil {
                return domain.MetricsSummary{}, err
        }
        m := s.Spec.Agent.Metrics
        return domain.ComputeMetrics(domain.MetricsInput{
                Events: events, MinutesSaved: m.MinutesSaved,
                BaselineHoursMin: m.MentorHoursBaselineMin, BaselineHoursMax: m.MentorHoursBaselineMax,
                Progress: progress, DocsRead: len(read),
                DocsTotal:     s.Services.Playbook.VisibleCount(s.Seed.Intern.Role, s.Seed.Intern.Team),
                OpenQuestions: len(open),
        }), nil
}

// Workspace возвращает состояние с учётом прав роли: переписку видит только стажёр,
// брифы — наставник и руководитель, инциденты безопасности — руководитель.
func (s *Service) Workspace(ctx context.Context, p domain.Principal, targetInternID ...string) (WorkspaceView, error) {
        var v WorkspaceView
        if err := s.authorize(ctx, p, OpWorkspaceRead); err != nil {
                return v, err
        }
        internID := s.resolveInternID(p, targetInternID...)
        intern, err := domain.LoadOne[domain.InternProfile](ctx, s.Store, domain.KindIntern, internID)
        if err != nil {
                return v, err
        }
        checklist, err := domain.LoadChecklist(ctx, s.Store, internID)
        if err != nil {
                return v, err
        }
        task, err := domain.LoadOne[domain.SprintTask](ctx, s.Store, domain.KindSprintTask, internID)
        if err != nil {
                return v, err
        }
        progress := domain.ComputeProgress(checklist)
        metrics, err := s.metrics(ctx, progress)
        if err != nil {
                return v, err
        }
        read, err := s.docsReadFor(ctx, internID)
        if err != nil {
                return v, err
        }
        notifications, err := domain.Load[domain.Notification](ctx, s.Store, domain.KindNotification, p.ID)
        if err != nil {
                return v, err
        }

        mode, model := s.Agent.Mode(ctx)
        v = WorkspaceView{
                Viewer:    ViewerInfo{ID: p.ID, Name: p.Name, Role: p.Role, RoleTitle: s.RBAC.RoleTitle(p.Role), Team: p.Team},
                Intern:    intern,
                Checklist: checklist, Progress: progress, NextStep: domain.NextStep(checklist),
                SprintTask: task, FeatureFlags: s.Services.Flags.List(ctx), Metrics: metrics,
                Messages: []domain.ChatMessage{}, Notifications: reverse(notifications),
                Briefs: []domain.MentorBrief{}, Absences: []domain.AbsenceRequest{},
                AccessGrants: []domain.AccessGrant{}, Incidents: []domain.TelemetryEvent{},
                DocProgress: DocProgress{Read: read, Total: metrics.DocsTotal},
                Engine:      EngineInfo{Mode: mode, Model: model},
                Storage:     StorageInfo{Store: s.Store.Name(), Cache: s.Cache.Name()},
                Permissions: map[string]bool{
                        "chat": s.can(p, OpChatSend), "cron": s.can(p, OpCronTrigger), "reset": s.can(p, OpReset),
                        "telemetry": s.can(p, OpTelemetryRead), "flags_write": s.can(p, OpFlagsWrite),
                        "briefs": s.can(p, OpBriefsRead), "incidents": s.can(p, OpIncidentsRead),
                },
                Disclaimer: s.Spec.Agent.Agent.SyntheticData.Disclaimer,
        }

        isOwner := p.ID == internID
        if s.can(p, OpChatHistoryRead) && isOwner {
                if v.Messages, err = domain.Load[domain.ChatMessage](ctx, s.Store, domain.KindMessage, internID); err != nil {
                        return v, err
                }
        }
        if isOwner || s.can(p, OpAbsencesTeam) {
                absences, err := domain.Load[domain.AbsenceRequest](ctx, s.Store, domain.KindAbsence, internID)
                if err != nil {
                        return v, err
                }
                grants, err := domain.Load[domain.AccessGrant](ctx, s.Store, domain.KindAccessGrant, internID)
                if err != nil {
                        return v, err
                }
                v.Absences, v.AccessGrants = reverse(absences), grants
        }
        if s.can(p, OpBriefsRead) {
                briefs, err := domain.Load[domain.MentorBrief](ctx, s.Store, domain.KindBrief, internID)
                if err != nil {
                        return v, err
                }
                v.Briefs = reverse(briefs)
        }
        if s.can(p, OpIncidentsRead) {
                events, err := domain.Load[domain.TelemetryEvent](ctx, s.Store, domain.KindEvent, internID)
                if err != nil {
                        return v, err
                }
                for i := len(events) - 1; i >= 0; i-- {
                        if events[i].Name == "rbac.incident" {
                                v.Incidents = append(v.Incidents, events[i])
                        }
                }
        }
        return v, nil
}

// ReviewAbsence согласовывает или отклоняет заявку на отгул руководителем.
func (s *Service) ReviewAbsence(ctx context.Context, p domain.Principal, id, decision, comment string) (*domain.AbsenceRequest, error) {
        if err := s.authorize(ctx, p, OpAbsencesReview); err != nil {
                return nil, err
        }
        id = strings.TrimSpace(id)
        decision = strings.TrimSpace(decision)
        comment = strings.TrimSpace(comment)
        if id == "" {
                return nil, fmt.Errorf("%w: не указан id заявки", domain.ErrValidation)
        }
        if decision != domain.AbsenceStatusApproved && decision != domain.AbsenceStatusRejected {
                return nil, fmt.Errorf("%w: некорректное решение %q (ожидается approved или rejected)", domain.ErrValidation, decision)
        }

        all, err := domain.Load[domain.AbsenceRequest](ctx, s.Store, domain.KindAbsence, "")
        if err != nil {
                return nil, err
        }
        var target *domain.AbsenceRequest
        for i := range all {
                if all[i].ID == id || all[i].Number == id {
                        target = &all[i]
                        break
                }
        }
        if target == nil {
                return nil, domain.ErrNotFound
        }

        now := s.Clock.Now()
        target.Status = decision
        target.Decision = decision
        target.Comment = comment
        target.ReviewerID = p.ID
        target.ReviewedAt = &now

        if err := domain.Save(ctx, s.Store, domain.KindAbsence, target.ID, target.InternID, target.CreatedAt, *target); err != nil {
                return nil, err
        }

        // 1. Clickstream-событие absence.reviewed
        s.Events.Track(ctx, target.InternID, p, domain.EventKindClickstream, "absence.reviewed", domain.SeverityInfo, map[string]any{
                "id":          target.ID,
                "number":      target.Number,
                "decision":    decision,
                "comment":     comment,
                "reviewer_id": p.ID,
        })

        // 2. Системное уведомление от AI-Buddy в чат стажёра
        var notifyText string
        if decision == domain.AbsenceStatusApproved {
                notifyText = fmt.Sprintf("✅ Руководитель согласовал твою заявку на отгул %s. Статус в системе «Отсутствия» обновлён.", target.Number)
        } else {
                notifyText = fmt.Sprintf("❌ Руководитель отклонил твою заявку на отгул %s. Статус в системе «Отсутствия» обновлён.", target.Number)
                if comment != "" {
                        notifyText += fmt.Sprintf("\nКомментарий: %s", comment)
                }
        }

        msg := domain.ChatMessage{
                ID:        s.IDs.NewID("msg"),
                InternID:  target.InternID,
                Role:      domain.MessageAssistant,
                Text:      notifyText,
                Intent:    "absence_reviewed",
                Engine:    domain.EngineSmart,
                Sources:   []domain.DocRef{},
                CreatedAt: now,
        }
        _ = s.saveMessage(ctx, msg)

        // Уведомление во входящие
        _, _ = s.Services.Notifications.Send(ctx, target.InternID, "Согласование отгула", notifyText, "absence", now)

        return target, nil
}

func reverse[T any](in []T) []T {
        out := make([]T, len(in))
        for i, v := range in {
                out[len(in)-1-i] = v
        }
        return out
}

// ChatResult — результат обработки сообщения или подтверждённого действия.
type ChatResult struct {
        UserMessage      *domain.ChatMessage    `json:"user_message"`
        Message          *domain.ChatMessage    `json:"message"`
        ToolCalls        []domain.MCPToolCall   `json:"tool_calls"`
        RBAC             *domain.RBACVerdict    `json:"rbac"`
        Checklist        []domain.ChecklistItem `json:"checklist"`
        ChecklistUpdates []domain.ChecklistItem `json:"checklist_updates"`
        Progress         domain.Progress        `json:"progress"`
        NextStep         *domain.ChecklistItem  `json:"next_step"`
        LatencyMS        float64                `json:"latency_ms"`
}

func (s *Service) saveMessage(ctx context.Context, m domain.ChatMessage) error {
        return domain.Save(ctx, s.Store, domain.KindMessage, m.ID, m.InternID, m.CreatedAt, m)
}

func callIDs(calls []domain.MCPToolCall) []string {
        ids := make([]string, 0, len(calls))
        for _, c := range calls {
                ids = append(ids, c.ID)
        }
        return ids
}

func nonNil[T any](in []T) []T {
        if in == nil {
                return []T{}
        }
        return in
}

// Chat обрабатывает сообщение стажёра: агент → MCP/RBAC → эффекты на чек-лист → ответ.
func (s *Service) Chat(ctx context.Context, p domain.Principal, text string, emit func(agent.Stage), targetInternID ...string) (ChatResult, error) {
        started := time.Now()
        var res ChatResult
        if err := s.authorize(ctx, p, OpChatSend); err != nil {
                return res, err
        }
        text = strings.TrimSpace(text)
        if n := len([]rune(text)); n == 0 || n > maxMessageLen {
                return res, fmt.Errorf("%w: сообщение должно быть от 1 до %d символов", domain.ErrValidation, maxMessageLen)
        }
        internID := s.resolveInternID(p, targetInternID...)
        intern, err := domain.LoadOne[domain.InternProfile](ctx, s.Store, domain.KindIntern, internID)
        if err != nil {
                return res, err
        }
        checklist, err := domain.LoadChecklist(ctx, s.Store, internID)
        if err != nil {
                return res, err
        }
        task, err := domain.LoadOne[domain.SprintTask](ctx, s.Store, domain.KindSprintTask, internID)
        if err != nil {
                return res, err
        }
        sess := s.loadSession(ctx, internID)

        user := domain.ChatMessage{
                ID: s.IDs.NewID("msg"), InternID: internID, Role: domain.MessageUser, Text: text,
                Sources: []domain.DocRef{}, Suggestions: []string{}, ToolCallIDs: []string{}, CreatedAt: s.Clock.Now(),
        }
        if err := s.saveMessage(ctx, user); err != nil {
                return res, err
        }
        s.Events.Track(ctx, internID, p, domain.EventKindClickstream, "chat.message_sent", domain.SeverityInfo,
                map[string]any{"length": len([]rune(text))})

        reply := s.Agent.Handle(ctx, agent.Request{
                Principal: p, Intern: intern, Text: text, Checklist: checklist, Task: task,
                LastIntent: sess.LastIntent, Emit: emit,
        })

        updates, err := s.applyEffects(ctx, p, reply.ToolCalls, reply.DocsRead, internID)
        if err != nil {
                return res, err
        }
        if reply.OpenQuestion != "" {
                q := openQuestion{Text: reply.OpenQuestion, AskedAt: s.Clock.Now()}
                if err := domain.Save(ctx, s.Store, domain.KindOpenQuestion, s.IDs.NewID("q"), internID, q.AskedAt, q); err != nil {
                        return res, err
                }
        }
        checklist, err = domain.LoadChecklist(ctx, s.Store, internID)
        if err != nil {
                return res, err
        }

        assistant := domain.ChatMessage{
                ID: s.IDs.NewID("msg"), InternID: internID, Role: domain.MessageAssistant,
                Text:   reply.Text + checklistNote(updates, checklist),
                Intent: reply.Intent, Engine: reply.Engine, Sources: nonNil(reply.Sources),
                ActionCard: reply.ActionCard, ResultCard: reply.ResultCard, Suggestions: nonNil(reply.Suggestions),
                ToolCallIDs: callIDs(reply.ToolCalls), CreatedAt: s.Clock.Now(),
        }
        if err := s.saveMessage(ctx, assistant); err != nil {
                return res, err
        }
        s.saveSession(ctx, internID, session{LastIntent: reply.Intent, Turns: sess.Turns + 1})

        latency := float64(time.Since(started).Microseconds()) / 1000
        payload := map[string]any{
                "intent": reply.Intent, "scenario": reply.Scenario, "engine": reply.Engine,
                "answered": reply.Answered, "latency_ms": latency, "tool_calls": len(reply.ToolCalls),
        }
        if reply.SavingKey != "" {
                payload["saving_key"] = reply.SavingKey
        }
        s.Events.Track(ctx, internID, p, domain.EventKindAgent, "agent.answer", domain.SeverityInfo, payload)
        if reply.LLMFallback != "" {
                s.Events.Track(ctx, internID, p, domain.EventKindAgent, "llm.fallback", domain.SeverityWarn,
                        map[string]any{"reason": reply.LLMFallback, "fallback": "smart_engine"})
        }

        return ChatResult{
                UserMessage: &user, Message: &assistant, ToolCalls: nonNil(reply.ToolCalls), RBAC: reply.Verdict,
                Checklist: checklist, ChecklistUpdates: nonNil(updates), Progress: domain.ComputeProgress(checklist),
                NextStep: domain.NextStep(checklist), LatencyMS: latency,
        }, nil
}

func checklistNote(updates, checklist []domain.ChecklistItem) string {
        if len(updates) == 0 {
                return ""
        }
        var b strings.Builder
        b.WriteString("\n\n")
        for _, u := range updates {
                state := "выполнен"
                if u.Status == domain.StatusInProgress {
                        state = "в работе"
                }
                fmt.Fprintf(&b, "Шаг чек-листа «%s» — %s. ", u.Title, state)
        }
        if next := domain.NextStep(checklist); next != nil {
                fmt.Fprintf(&b, "Следующий шаг: «%s».", next.Title)
        } else {
                b.WriteString("Все шаги чек-листа выполнены!")
        }
        return strings.TrimRight(b.String(), " ")
}

func evidence(call domain.MCPToolCall) string {
        switch call.Tool {
        case mcp.ToolRequestRepoAccess:
                return fmt.Sprintf("Доступ к %v выдан через IAM (AI-Buddy)", call.Result["repository"])
        case mcp.ToolCreateDayOff:
                return fmt.Sprintf("Заявка %v оформлена через AI-Buddy", call.Result["request_number"])
        case mcp.ToolSendBrief:
                return "Бриф к 1:1 отправлен наставнику"
        }
        return "Выполнено через AI-Buddy"
}

// applyEffects применяет правила checklist_effects из спецификации агента.
func (s *Service) applyEffects(ctx context.Context, p domain.Principal, calls []domain.MCPToolCall, docs []string, targetInternID ...string) ([]domain.ChecklistItem, error) {
        internID := s.resolveInternID(p, targetInternID...)
        now := s.Clock.Now()
        for _, id := range docs {
                if err := domain.Save(ctx, s.Store, domain.KindDocRead, internID+":"+id, internID, now, docRead{DocID: id, ReadAt: now}); err != nil {
                        return nil, err
                }
        }
        read := map[string]bool{}
        if len(docs) > 0 {
                ids, err := s.docsReadFor(ctx, internID)
                if err != nil {
                        return nil, err
                }
                for _, id := range ids {
                        read[id] = true
                }
        }

        var updates []domain.ChecklistItem
        for _, eff := range s.Spec.Agent.ChecklistEffects {
                status, proof := domain.ChecklistStatus(""), ""
                switch {
                case eff.Tool != "":
                        for _, c := range calls {
                                if c.Tool == eff.Tool && c.OK() {
                                        status, proof = domain.ChecklistStatus(eff.Status), evidence(c)
                                }
                        }
                case len(eff.DocsRead) > 0 && len(docs) > 0:
                        done := 0
                        for _, id := range eff.DocsRead {
                                if read[id] {
                                        done++
                                }
                        }
                        switch {
                        case done == len(eff.DocsRead):
                                status, proof = domain.ChecklistStatus(eff.Status), "Инструкции изучены через AI-Buddy"
                        case done > 0:
                                status, proof = domain.StatusInProgress, fmt.Sprintf("Изучено инструкций: %d из %d", done, len(eff.DocsRead))
                        }
                }
                if status == "" {
                        continue
                }
                item, err := domain.LoadOne[domain.ChecklistItem](ctx, s.Store, domain.KindChecklist, domain.ChecklistRecordID(internID, eff.Item))
                if errors.Is(err, domain.ErrNotFound) {
                        continue
                }
                if err != nil {
                        return nil, err
                }
                next, changed, err := item.Advance(status, proof, now)
                if err != nil || !changed {
                        continue // шаг уже в этом или более позднем статусе
                }
                if err := domain.SaveChecklistItem(ctx, s.Store, internID, next); err != nil {
                        return nil, err
                }
                updates = append(updates, next)
                s.Events.Track(ctx, internID, p, domain.EventKindClickstream, "checklist.updated", domain.SeverityInfo,
                        map[string]any{"item": next.ID, "status": string(next.Status), "evidence": next.Evidence})
        }
        return updates, nil
}
